import XCTest

// Include LearningLabJr.storekit in the UI-test bundle when activating these fixtures.

@MainActor
final class ReleasePaywallUITests: XCTestCase {
    func testParentGatePaywallDisclosureAndPurchase() async throws {
        continueAfterFailure = false
        let store = try makeStoreSession()
        store.resetToDefaultState()
        store.clearTransactions()
        store.disableDialogs = true
        defer { store.clearTransactions() }
        let app = XCUIApplication()
        app.launchArguments = ["-parents.voicePromptsEnabled", "NO", "-parents.backgroundMusicEnabled", "NO", "-parents.familyPlaySession.v1", ""]
        app.launch()
        let category = app.buttons["category.ABCs & Phonics"]
        XCTAssertTrue(category.waitForExistence(timeout: 15))
        category.tap()
        dismissLaunchPrompt(app)
        let letterDraw = app.buttons.matching(NSPredicate(format: "label BEGINSWITH 'Letter Draw.'")).firstMatch
        XCTAssertTrue(letterDraw.waitForExistence(timeout: 5))
        letterDraw.tap()
        XCTAssertTrue(app.buttons["premium.subscribe"].waitForExistence(timeout: 15))
        XCTAssertTrue(app.buttons["premium.plan.annual"].exists)
        XCTAssertTrue(app.buttons["premium.subscribe"].label.contains("29.99"))
        XCTAssertFalse(app.staticTexts["premium.offer"].exists)
        XCTAssertTrue(app.buttons["premium.plan.annual"].label.contains("SAVE 37%"))
        for _ in 0..<3 {
            if app.buttons["premium.plan.monthly"].isHittable { break }
            app.swipeUp()
        }
        app.buttons["premium.plan.monthly"].tap()
        XCTAssertTrue(app.buttons["premium.subscribe"].label.contains("3.99"))
        XCTAssertTrue(app.staticTexts["premium.renewal"].label.contains("month"))
        app.buttons["premium.plan.annual"].tap()
        XCTAssertTrue(app.buttons["premium.subscribe"].label.contains("29.99"))
        let plans = XCTAttachment(screenshot: app.screenshot())
        plans.name = "Premium Monthly and Annual Plans"
        plans.lifetime = .keepAlways
        add(plans)
        for _ in 0..<4 {
            if app.buttons["Terms of Use"].isHittable { break }
            app.swipeUp()
        }
        XCTAssertTrue(app.buttons["Privacy Policy"].exists)
        XCTAssertTrue(app.buttons["Terms of Use"].exists)
        XCTAssertTrue(app.buttons["Restore Purchases"].exists)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Subscription disclosure and legal links"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        for _ in 0..<4 {
            if app.buttons["premium.subscribe"].isHittable { break }
            app.swipeDown()
        }
        app.buttons["premium.subscribe"].tap()
        let question = app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH 'Please add '")).firstMatch
        XCTAssertTrue(question.waitForExistence(timeout: 5))
        let numbers = question.label.split(whereSeparator: { !$0.isNumber }).compactMap { Int($0) }
        app.textFields["Your answer"].tap()
        app.textFields["Your answer"].typeText(String(numbers.reduce(0, +)))
        app.buttons["Continue"].tap()
        XCTAssertTrue(letterDraw.waitForExistence(timeout: 15))
        XCTAssertFalse(letterDraw.label.contains("Premium, grown-up required"))
        letterDraw.tap()
        XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
    }
}

@MainActor
final class CategoryAccessUITests: XCTestCase {
    private let sections: [(String, String, String)] = [
        ("ABCs & Phonics", "phonics.letter-match", "phonics.letter-draw"),
        ("Shapes & Colors", "shapes.post", "shapes.laundry"),
        ("123s & Counting", "counting.pile-match", "counting.touch-count"),
        ("Nature Explorers", "nature.habitats", "nature.tracks"),
        ("Story Time", "stories.library", "stories.pictureHunt"),
        ("Big Feelings", "feelings.faces", "feelings.breathe")
    ]

    func testFreeFirstGameAndLockedSecondGameAcrossAllSections() throws {
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        for (section, first, second) in sections {
            let app = launch(section)
            let free = app.buttons["activity.\(first)"]
            XCTAssertTrue(free.waitForExistence(timeout: 5))
            XCTAssertFalse(free.label.contains("Premium, grown-up required"))
            free.tap()
            XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
            app.navigationBars.buttons.firstMatch.tap()
            let locked = app.buttons["activity.\(second)"]
            XCTAssertTrue(locked.waitForExistence(timeout: 5))
            XCTAssertTrue(locked.label.contains("Premium, grown-up required"))
            locked.tap()
            tapPaywallButton(app, "premium.subscribe")
            XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 5))
            XCTAssertFalse(app.buttons["premium.subscribe"].exists)
            app.buttons["Back to play"].tap()
            app.terminate()
        }
    }

    func testPremiumUnlocksEverySectionAndRevocationClosesOpenGame() async throws {
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        let transaction = try await store.buyProduct(identifier: "com.learninglabjr.premium.monthly")
        for (section, _, second) in sections {
            let app = launch(section)
            let paid = app.buttons["activity.\(second)"]
            XCTAssertTrue(paid.waitForExistence(timeout: 5))
            let unlocked = XCTNSPredicateExpectation(predicate: NSPredicate(format: "NOT label CONTAINS %@", "Premium, grown-up required"), object: paid)
            XCTAssertEqual(XCTWaiter.wait(for: [unlocked], timeout: 10), .completed)
            paid.tap()
            XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
            if section == "Big Feelings" {
                try store.refundTransaction(identifier: UInt(transaction.id))
                XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 10))
                XCTAssertFalse(app.buttons["premium.subscribe"].exists)
            }
            app.terminate()
        }
    }

    private func launch(_ section: String) -> XCUIApplication {
        let app = XCUIApplication()
        app.launchArguments = ["-parents.voicePromptsEnabled", "NO", "-parents.backgroundMusicEnabled", "NO", "-parents.familyPlaySession.v1", ""]
        app.launch()
        let category = app.buttons["category.\(section)"]
        XCTAssertTrue(category.waitForExistence(timeout: 15))
        for _ in 0..<3 {
            if category.isHittable { break }
            app.swipeUp()
        }
        category.tap()
        dismissLaunchPrompt(app)
        return app
    }
}

@MainActor
final class BookAccessUITests: XCTestCase {
    func testFirstBookFreeOtherBooksLockedAndPremiumUnlocksThem() async throws {
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        let app = XCUIApplication()
        app.launchArguments = ["-parents.voicePromptsEnabled", "NO", "-parents.backgroundMusicEnabled", "NO", "-parents.familyPlaySession.v1", ""]
        app.launch()
        func tap(_ element: XCUIElement) {
            for _ in 0..<6 {
                if element.exists && element.isHittable { element.tap(); return }
                app.swipeUp()
            }
            XCTFail("Missing control: \(element)")
        }
        XCTAssertTrue(app.buttons["category.Story Time"].waitForExistence(timeout: 15))
        tap(app.buttons["category.Story Time"])
        dismissLaunchPrompt(app)
        tap(app.buttons["activity.stories.library"])
        let free = app.buttons["activity.books.owen-onion"]
        XCTAssertTrue(free.waitForExistence(timeout: 5))
        XCTAssertFalse(free.label.contains("Premium, grown-up required"))
        tap(free)
        XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
        app.navigationBars.buttons.firstMatch.tap()
        let paidIDs = ["books.dino-basketball", "books.dino-hockey", "books.dino-baseball"]
        for id in paidIDs {
            let book = app.buttons["activity.\(id)"]
            XCTAssertTrue(book.label.contains("Premium, grown-up required"))
            tap(book)
            tapPaywallButton(app, "premium.subscribe")
            XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 5))
            XCTAssertFalse(app.buttons["premium.subscribe"].exists)
            app.buttons["Back to play"].tap()
            app.buttons["premium.close"].tap()
        }
        let transaction = try await store.buyProduct(identifier: "com.learninglabjr.premium.monthly")
        for id in paidIDs {
            let book = app.buttons["activity.\(id)"]
            let unlocked = XCTNSPredicateExpectation(predicate: NSPredicate(format: "NOT label CONTAINS %@", "Premium, grown-up required"), object: book)
            XCTAssertEqual(XCTWaiter.wait(for: [unlocked], timeout: 10), .completed)
            tap(book)
            XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
            if id == paidIDs.last {
                try store.refundTransaction(identifier: UInt(transaction.id))
                XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 10))
                app.buttons["Back to play"].tap()
            } else {
                app.navigationBars.buttons.firstMatch.tap()
            }
        }
        XCTAssertTrue(free.waitForExistence(timeout: 5))
        XCTAssertFalse(free.label.contains("Premium, grown-up required"))
        tap(free)
        XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
    }
}

@MainActor
private func dismissLaunchPrompt(_ app: XCUIApplication) {
    let close = app.buttons["premium.close"]
    if close.waitForExistence(timeout: 3) { close.tap() }
}

@MainActor
final class LaunchSubscriptionPromptUITests: XCTestCase {
    private func launch() -> XCUIApplication {
        let app = XCUIApplication()
        app.launchArguments = ["-parents.voicePromptsEnabled", "NO", "-parents.backgroundMusicEnabled", "NO", "-parents.familyPlaySession.v1", ""]
        app.launch()
        XCTAssertTrue(app.buttons["category.ABCs & Phonics"].waitForExistence(timeout: 15))
        return app
    }

    func testFirstTapCanCloseAndPlayFreeGamesWhileLockedGamesPromptAgain() throws {
        continueAfterFailure = false
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        for closeID in ["premium.close", "premium.continueFree"] {
            let app = launch()
            XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
            app.buttons["category.ABCs & Phonics"].tap()
            XCTAssertTrue(app.buttons["premium.plan.annual"].waitForExistence(timeout: 10))
            XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
            tapPaywallButton(app, closeID)
            let free = app.buttons["activity.phonics.letter-match"]
            XCTAssertTrue(free.waitForExistence(timeout: 5))
            free.tap()
            XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
            app.navigationBars.buttons.firstMatch.tap()
            for _ in 0..<2 {
                app.buttons["activity.phonics.letter-draw"].tap()
                XCTAssertTrue(app.buttons["premium.plan.annual"].waitForExistence(timeout: 5))
                app.buttons["premium.close"].tap()
                XCTAssertTrue(free.waitForExistence(timeout: 5))
            }
            app.navigationBars.buttons.firstMatch.tap()
            app.buttons["category.ABCs & Phonics"].tap()
            XCTAssertTrue(free.waitForExistence(timeout: 5))
            XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
            app.terminate()
        }
    }

    func testExistingMonthlyAndAnnualSubscribersDoNotSeeFirstTapPrompt() async throws {
        continueAfterFailure = false
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        for id in ["com.learninglabjr.premium.monthly", "com.learninglabjr.premium.annual"] {
            store.clearTransactions()
            _ = try await store.buyProduct(identifier: id)
            let app = launch()
            app.buttons["category.ABCs & Phonics"].tap()
            let paid = app.buttons["activity.phonics.letter-draw"]
            XCTAssertTrue(paid.waitForExistence(timeout: 10))
            XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
            XCTAssertFalse(paid.label.contains("Premium, grown-up required"))
            paid.tap()
            XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
            XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
            app.terminate()
        }
    }

    func testFirstTapOnHomeArtworkAlsoPrompts() throws {
        continueAfterFailure = false
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        XCUIDevice.shared.orientation = .portrait
        let app = launch()
        app.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.17)).tap()
        XCTAssertTrue(app.buttons["premium.close"].waitForExistence(timeout: 10))
        let shot = XCTAttachment(screenshot: app.screenshot())
        shot.name = "First-tap subscription invitation"
        shot.lifetime = .keepAlways
        add(shot)
        app.buttons["premium.close"].tap()
        app.buttons["category.ABCs & Phonics"].tap()
        XCTAssertTrue(app.buttons["activity.phonics.letter-match"].waitForExistence(timeout: 5))
        XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
    }

    func testCancelingParentCheckKeepsPlansAndDoesNotPurchase() throws {
        continueAfterFailure = false
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions() }
        let app = launch()
        app.buttons["category.ABCs & Phonics"].tap()
        XCTAssertTrue(app.buttons["premium.plan.annual"].waitForExistence(timeout: 10))
        XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists)
        tapPaywallButton(app, "premium.plan.monthly")
        for action in ["premium.subscribe", "Restore Purchases", "Privacy Policy"] {
            tapPaywallButton(app, action)
            XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 5))
            XCTAssertTrue(store.allTransactions().isEmpty)
            app.buttons["Back to play"].tap()
            XCTAssertTrue(app.buttons["premium.close"].waitForExistence(timeout: 5))
        }
        XCTAssertTrue(app.buttons["premium.subscribe"].label.contains("3.99"))
        app.buttons["premium.close"].tap()
        XCTAssertTrue(app.buttons["activity.phonics.letter-match"].waitForExistence(timeout: 5))
        XCTAssertTrue(store.allTransactions().isEmpty)
    }

    func testParentsButtonAlsoPromptsOnFirstTapInLandscape() throws {
        continueAfterFailure = false
        let store = try makeStoreSession()
        store.resetToDefaultState(); store.clearTransactions(); store.disableDialogs = true
        defer { store.clearTransactions(); XCUIDevice.shared.orientation = .portrait }
        XCUIDevice.shared.orientation = .landscapeLeft
        let app = launch()
        app.buttons["home.parents"].tap()
        XCTAssertTrue(app.buttons["premium.close"].waitForExistence(timeout: 10))
        app.buttons["premium.close"].tap()
        XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 5))
        XCTAssertFalse(app.buttons["premium.plan.annual"].exists)
    }
}

@MainActor
private func makeStoreSession() throws -> SKTestSession {
    let url = try XCTUnwrap(Bundle(for: ReleasePaywallUITests.self).url(forResource: "LearningLabJr", withExtension: "storekit"))
    return try SKTestSession(contentsOf: url)
}

@MainActor
private func tapPaywallButton(_ app: XCUIApplication, _ identifier: String) {
    let button = app.buttons[identifier]
    for _ in 0..<5 {
        if button.exists && button.isHittable { button.tap(); return }
        app.swipeUp()
    }
    for _ in 0..<5 {
        if button.exists && button.isHittable { button.tap(); return }
        app.swipeDown()
    }
    XCTFail("Missing paywall button: \(identifier)")
}
