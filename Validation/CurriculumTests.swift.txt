    @MainActor
    func testWordBuilderBankRecentWindowAndRuth() {
        let bank = WordBuilderWord.bank
        XCTAssertGreaterThan(bank.count, 20)
        XCTAssertEqual(Set(bank.map(\.word)).count, bank.count)
        XCTAssertEqual(Set(bank.map { $0.letters.count }), [3, 4, 5])
        XCTAssertTrue(bank.contains { Set($0.letters).count < $0.letters.count })
        for item in bank {
            XCTAssertNotNil(UIImage(named: item.asset) ?? UIImage(named: "Space/\(item.asset)"), item.asset)
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: item.prompt), item.prompt)
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: item.success), item.success)
            for index in item.letters.indices {
                let letter = item.letters[index]
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: "Look at the model. Find \(letter) next."))
                if index + 1 < item.letters.count {
                    XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: "\(letter). Next find \(item.letters[index + 1])."))
                }
            }
        }
        var recent: [String] = []
        for _ in 0..<2000 {
            let next = WordBuilderWord.next(recent: recent)
            XCTAssertFalse(recent.contains(next.word))
            recent = WordBuilderWord.remembering(next.word, recent: recent)
            XCTAssertLessThanOrEqual(recent.count, 20)
            XCTAssertEqual(recent.last, next.word)
            // Match persisted history restored on the next game entry.
            recent = recent.joined(separator: "|").split(separator: "|").map(String.init)
        }
        let old = Array(bank.prefix(21).map(\.word))
        XCTAssertEqual(WordBuilderWord.remembering("CAT", recent: old), Array((old + ["CAT"]).suffix(20)))
    }

    @MainActor
    func testBeginningSoundsAlphabetPicturesAndRuthCoverage() {
        let alphabet = Set(LetterDrawAlphabet.letters)
        XCTAssertEqual(BeginningSoundsPair.alphabet.count, 26)
        for pair in BeginningSoundsPair.alphabet {
            XCTAssertNotEqual(pair.word, pair.matchingWord)
            for word in [pair.word, pair.matchingWord] {
                XCTAssertEqual(String(word.prefix(1)).uppercased(), pair.letter)
            }
            for asset in [pair.asset, pair.matchingAsset].compactMap({ $0 }) {
                XCTAssertNotNil(UIImage(named: asset) ?? UIImage(named: "Space/\(asset)"), asset)
            }
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: pair.success), pair.success)
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: pair.retry), pair.retry)
        }
        for previous in LetterDrawAlphabet.letters {
            let rounds = BeginningSoundsSessionRound.session(previousStart: previous, lastShown: "Z")
            XCTAssertEqual(rounds.count, 26)
            XCTAssertEqual(Set(rounds.map { $0.pair.letter }), alphabet)
            let distractors = rounds.compactMap { round in round.choices.first { $0.word != round.pair.matchingWord }?.word }
            XCTAssertEqual(Set(distractors).count, 26)
            XCTAssertNotEqual(rounds.first?.pair.letter, previous)
            XCTAssertNotEqual(rounds.first?.pair.letter, "Z")
            for round in rounds {
                XCTAssertEqual(round.choices.count, 2)
                XCTAssertEqual(round.choices.filter { $0.word == round.pair.matchingWord }.count, 1)
                XCTAssertEqual(round.choices.filter { String($0.word.prefix(1)).uppercased() == round.pair.letter }.count, 1)
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: round.prompt), round.prompt)
                let reversed = BeginningSoundsSessionRound(pair: round.pair, choices: round.choices.reversed())
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: reversed.prompt), reversed.prompt)
            }
        }
    }

    @MainActor
    func testSoundBasketsAlphabetAndDropMatching() {
        let alphabet = Set(LetterDrawAlphabet.letters)
        XCTAssertEqual(SoundBasketItem.alphabet.count, 26)
        XCTAssertEqual(Set(SoundBasketItem.alphabet.map(\.letter)), alphabet)
        for item in SoundBasketItem.alphabet {
            XCTAssertEqual(String(item.word.prefix(1)).uppercased(), item.letter)
            if let asset = item.asset {
                XCTAssertNotNil(UIImage(named: asset) ?? UIImage(named: "Space/\(asset)"), asset)
            }
        }
        for previous in LetterDrawAlphabet.letters {
            let rounds = SoundBasketRound.session(previousStart: previous, lastShown: "Z")
            XCTAssertEqual(rounds.count, 26)
            XCTAssertEqual(Set(rounds.map { $0.item.letter }), alphabet)
            XCTAssertNotEqual(rounds.first?.item.letter, previous)
            XCTAssertNotEqual(rounds.first?.item.letter, "Z")
            for round in rounds {
                XCTAssertEqual(Set(round.choices).count, 2)
                XCTAssertTrue(round.choices.contains(round.item.letter))
                XCTAssertTrue(Set(round.choices).isSubset(of: alphabet))
                XCTAssertTrue(round.matches([round.dragID], basket: round.item.letter))
                XCTAssertFalse(round.matches(["stale-picture"], basket: round.item.letter))
                let wrong = round.choices.first { $0 != round.item.letter }!
                XCTAssertFalse(round.matches([round.dragID], basket: wrong))
            }
        }
    }

    @MainActor
    func testLetterDrawUsesEveryLetterOnceAndRestartsWithDifferentFirst() {
        for previous in [""] + LetterDrawAlphabet.letters {
            let letters = LetterDrawAlphabet.shuffled(startingAfter: previous, lastShown: "Z")
            XCTAssertEqual(letters.count, 26)
            XCTAssertEqual(Set(letters), Set(LetterDrawAlphabet.letters))
            XCTAssertNotEqual(letters.first, previous)
            XCTAssertNotEqual(letters.first, "Z")
            let play = LiteracyPlay(total: letters.count)
            var seen: Set<String> = []
            for index in 0..<26 {
                XCTAssertFalse(play.complete)
                XCTAssertEqual(play.round, index)
                XCTAssertTrue(seen.insert(letters[play.round]).inserted)
                play.advance()
                XCTAssertEqual(play.round, index, "Unfinished letters cannot be skipped")
                play.win("Done")
                play.advance()
            }
            XCTAssertTrue(play.complete)
            play.advance()
            XCTAssertEqual(play.round, 25)
            play.replay()
            XCTAssertEqual(play.round, 0)
            XCTAssertFalse(play.solved)
            XCTAssertFalse(play.complete)
        }
        XCTAssertEqual(LiteracyPlay().total, 3, "Other literacy games retain their round count")
    }

    @MainActor
    func testRecordedNarrationNormalizesTextAndRejectsUnsafePaths() {
        let good = NarrationAudioCatalog.Clip(text: "Hi,\n friend!", file: "llj-example.mp3", spokenText: "Hi, friend!", voice: "Joanna", engine: "generative", category: "shared")
        let unsafe = NarrationAudioCatalog.Clip(text: "Unsafe", file: "llj-../../other.mp3", spokenText: "Unsafe", voice: "Joanna", engine: "generative", category: "shared")
        let catalog = NarrationAudioCatalog(manifest: .init(schemaVersion: 1, clips: [good, unsafe]))
        XCTAssertEqual(catalog.clip(for: "  Hi,  friend! ")?.file, good.file)
        XCTAssertNil(catalog.clip(for: "Unsafe"))
        XCTAssertNil(catalog.url(for: "A prompt that has no recording"))
        XCTAssertEqual(NarrationAudioCatalog.normalized("caf\u{65}\u{301}"), "café")
        XCTAssertEqual(NarrationAudioCatalog(manifest: .init(schemaVersion: 99, clips: [good])).count, 0)
    }

    @MainActor
    func testBundledPollyNarrationIsPlayableAndCoversSharedText() throws {
        let catalog = NarrationAudioCatalog.shared
        XCTAssertGreaterThan(catalog.count, 500)
        XCTAssertNotNil(catalog.url(for: NarrationVoice.previewText))
        XCTAssertNotNil(catalog.url(for: "We did it together! You can play again or choose all done."))
        let manifestURL = try XCTUnwrap(Bundle.main.url(forResource: "NarrationAudioManifest", withExtension: "json"))
        let manifest = try JSONDecoder().decode(NarrationAudioCatalog.Manifest.self, from: Data(contentsOf: manifestURL))
        XCTAssertEqual(Set(manifest.clips.map { NarrationAudioCatalog.normalized($0.text) }).count, manifest.clips.count)
        XCTAssertEqual(Set(manifest.clips.map(\.voice)), ["Ruth"])
        for clip in manifest.clips {
            let url = try XCTUnwrap(catalog.url(for: clip.text), clip.text)
            let audio = try AVAudioPlayer(contentsOf: url)
            XCTAssertGreaterThan(audio.duration, 0.1, clip.text)
        }
    }

    func testSessionClockExpiryAndBackgroundElapsedTime() {
        var clock = PlaySessionClock()
        XCTAssertEqual(clock.remaining(at: 100), 0)
        clock.start(minutes: 5, now: 100)
        XCTAssertEqual(clock.remaining(at: 100), 300)
        XCTAssertEqual(clock.remaining(at: 340), 60)
        XCTAssertEqual(clock.remaining(at: 399.1), 1)
        XCTAssertEqual(clock.remaining(at: 400), 0)
        XCTAssertEqual(clock.remaining(at: 5000), 0)
    }

    func testSessionRestartAndStopCannotRetainPreviousDeadline() {
        var clock = PlaySessionClock()
        clock.start(minutes: 30, now: 0)
        clock.start(minutes: 1, now: 20)
        XCTAssertEqual(clock.remaining(at: 21), 59)
        clock.stop()
        XCTAssertNil(clock.deadline)
        XCTAssertEqual(clock.remaining(at: 22), 0)
        clock.start(minutes: Int.max, now: 40)
        XCTAssertEqual(clock.remaining(at: 40), 1800)
        clock.start(minutes: Int.min, now: 40)
        XCTAssertEqual(clock.remaining(at: 40), 60)
    }

    @MainActor
    func testNarrationPrefersNaturalQualityAndHonorsParentSelection() {
        let standardUS = NarrationVoiceDescriptor(id: "standard", name: "Standard", language: "en-US", quality: 1, gender: .female)
        let premiumUK = NarrationVoiceDescriptor(id: "premium-uk", name: "Premium UK", language: "en-GB", quality: 3, gender: .female)
        let premiumUS = NarrationVoiceDescriptor(id: "premium-us", name: "Premium US", language: "en-US", quality: 3, gender: .female)
        let voices = [standardUS, premiumUK, premiumUS]
        XCTAssertEqual(NarrationVoiceSelection.selectedIdentifier(in: voices, preferredIdentifier: ""), premiumUS.id)
        XCTAssertEqual(NarrationVoiceSelection.selectedIdentifier(in: [standardUS, premiumUK], preferredIdentifier: ""), premiumUK.id)
        XCTAssertEqual(NarrationVoiceSelection.selectedIdentifier(in: voices, preferredIdentifier: standardUS.id), standardUS.id)
        XCTAssertEqual(NarrationVoiceSelection.selectedIdentifier(in: voices, preferredIdentifier: "deleted-voice"), premiumUS.id)
        XCTAssertEqual(NarrationVoiceSelection.ranked(voices), NarrationVoiceSelection.ranked(voices.reversed()))
    }

    @MainActor
    func testNarrationExcludesNoveltyPersonalAndNonEnglishVoices() {
        let unsuitable = [
            NarrationVoiceDescriptor(id: "novelty", name: "Bubbles", language: "en-US", quality: 3, gender: .unspecified),
            NarrationVoiceDescriptor(id: "flagged", name: "Character", language: "en-US", quality: 3, gender: .female, isNovelty: true),
            NarrationVoiceDescriptor(id: "personal", name: "Personal", language: "en-US", quality: 3, gender: .female, isPersonal: true),
            NarrationVoiceDescriptor(id: "french", name: "French", language: "fr-FR", quality: 3, gender: .female)
        ]
        XCTAssertTrue(NarrationVoiceSelection.ranked(unsuitable).isEmpty)
        XCTAssertNil(NarrationVoiceSelection.selectedIdentifier(in: unsuitable, preferredIdentifier: "personal"))
        XCTAssertNil(NarrationVoiceSelection.selectedIdentifier(in: [], preferredIdentifier: ""))
    }
