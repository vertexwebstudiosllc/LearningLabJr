                let context = try XCTUnwrap(CGContext(data: bytes.baseAddress, width: 32, height: 32,
                    bitsPerComponent: 8, bytesPerRow: 128, space: CGColorSpaceCreateDeviceRGB(),
                    bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue))
                context.draw(image, in: CGRect(x: 0, y: 0, width: 32, height: 32))

            XCTAssertGreaterThan(alpha.filter { $0 < 10 }.count, 100, name)
            XCTAssertGreaterThan(alpha.filter { $0 > 240 }.count, 100, name)
            for corner in [0, 31, 992, 1023] { XCTAssertLessThan(alpha[corner], 10, name) }
        }

    func testDinosaurTrailsPairSpacingProgressionAndRuth() {
        var recent: [String] = []
        for dino in TrailDinosaur.bank {
            XCTAssertNotNil(UIImage(named: dino.id), dino.id)
            XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: dino.name)?.voice, "Ruth", dino.name)
        }
        for text in [DinosaurTrailSession.stepHint, DinosaurTrailSession.completion] {

        }
        for _ in 0..<100 {
            var play = DinosaurTrailSession(recentPairs: recent)
            XCTAssertEqual(play.rounds.count, 15)
            XCTAssertEqual(Set(play.rounds.map(\.pairID)).count, 15)
            for index in 0..<15 {
                let round = play.current
                XCTAssertEqual(round.steps, 4 + (index / 5) * 2)
                XCTAssertEqual(Set(round.dinosaurs.map(\.id)).count, 2)
                XCTAssertFalse(recent.suffix(5).contains(round.pairID))
                recent.append(round.pairID)
                for text in [round.prompt, round.success, round.retry] {
                    XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)
                    XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: text)?.voice, "Ruth", text)
                }
                for bends in round.bends {
                    XCTAssertEqual(bends.count, round.steps)
                    XCTAssertEqual(bends.first, 0.5); XCTAssertEqual(bends.last, 0.5)
                    XCTAssertTrue(bends.allSatisfy { (0.12...0.88).contains($0) })
                }
                play.next(); XCTAssertEqual(play.index, index)
                XCTAssertFalse(play.step(trail: -1, footprint: 0))
                XCTAssertFalse(play.step(trail: 2, footprint: 0))
                XCTAssertFalse(play.step(trail: 0, footprint: -1))
                for trail in [1 - round.target, round.target] {
                    for step in 0..<round.steps {
                        XCTAssertFalse(play.step(trail: trail, footprint: step + 1))
                        XCTAssertTrue(play.step(trail: trail, footprint: step))
                        XCTAssertFalse(play.step(trail: trail, footprint: step))
                        XCTAssertEqual(play.progress[trail], step + 1)
                    }
                    if trail != round.target {
                        XCTAssertFalse(play.solved)
                        play.next(); XCTAssertEqual(play.index, index)
                    }
                }
                XCTAssertTrue(play.solved)
                play.next()
            }
            XCTAssertTrue(play.complete)
            play.next(); XCTAssertEqual(play.index, 15)
            XCTAssertFalse(play.step(trail: 0, footprint: 0))
            recent = Array(recent.suffix(5))
        }
    }

    @MainActor
    func testHabitatGroupsNoRepeatsProgressionAndRuth() {
        var previous: String?
        var signatures: Set<String> = []
        for animal in HabitatAnimal.bank {
            if let asset = animal.asset { XCTAssertNotNil(UIImage(named: asset), asset) }
            for text in [animal.prompt, animal.hint] {
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)
                XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: text)?.voice, "Ruth")
            }
        }
        for text in [HabitatSession.groupPrompt, HabitatSession.completion, "Choose an animal. Then choose its home."] {
            XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: text)?.voice, "Ruth", text)
        }
        for _ in 0..<100 {
            var play = HabitatSession(previousFirst: previous)
            XCTAssertEqual(play.rounds.count, 6)
            XCTAssertNotEqual(play.selected, previous)
            previous = play.selected
            let all = play.rounds.flatMap(\.animals)
            XCTAssertEqual(all.count, 24)
            XCTAssertEqual(Set(all.map(\.id)).count, 24)
            signatures.insert(all.map(\.id).joined(separator: ","))
            XCTAssertEqual(Set(play.rounds.map { $0.homes.map(\.rawValue).sorted().joined(separator: ",") }).count, 6)
            for index in 0..<6 {
                XCTAssertEqual(play.index, index)
                XCTAssertFalse(play.complete)
                play.next(); XCTAssertEqual(play.index, index)
                XCTAssertEqual(Set(play.current.homes).count, 2)
                let pair = Set(play.current.homes)
                XCTAssertNotEqual(pair, Set([AnimalHabitat.trees, .rainforest]))
                XCTAssertNotEqual(pair, Set([AnimalHabitat.farm, .pond]))
                if pair.contains(.garden) { XCTAssertTrue(pair.contains(.farm) || pair.contains(.ocean)) }
                for home in play.current.homes {
                    XCTAssertEqual(play.current.animals.filter { $0.habitat == home }.count, 2)
                }
                for animal in play.current.animals.reversed() {
                    play.select(animal.id)
                    let count = play.matched.count
                    for wrong in AnimalHabitat.allCases where wrong != animal.habitat {
                        XCTAssertFalse(play.place(in: wrong))
                        XCTAssertEqual(play.matched.count, count)
                    }
                    XCTAssertTrue(play.place(in: animal.habitat))
                    XCTAssertEqual(play.matched.count, count + 1)
                    let selected = play.selected
                    play.select(animal.id); XCTAssertEqual(play.selected, selected)
                    play.select("invalid"); XCTAssertEqual(play.selected, selected)
                }
                XCTAssertTrue(play.solved)
                XCTAssertFalse(play.place(in: play.current.homes[0]))
                play.next()
            }
            XCTAssertTrue(play.complete)
            play.next(); XCTAssertEqual(play.index, 6)
            play.select("cow"); XCTAssertNil(play.selected)
            XCTAssertFalse(play.place(in: .farm))
        }
        XCTAssertEqual(signatures.count, 100)
    }

    @MainActor
    func testRemainingCountingGamesProgressionAndNarration() async {
        let totals: [CountingActivity: Int] = [.share: 10, .trail: 20, .more: 20, .tickets: 24]
        var narration: Set<String> = []
        func record(_ play: CountingPlay) { if !play.feedback.isEmpty { narration.insert(play.feedback) } }
        for kind in CountingActivity.allCases {
            let play = CountingPlay(kind: kind)
            XCTAssertEqual(play.total, totals[kind])
            narration.insert(kind.completion)
            for index in 0..<play.total {
                XCTAssertFalse(play.complete)
                XCTAssertFalse(play.solved)
                XCTAssertEqual(play.round, index)
                narration.insert(play.prompt)
                let n = play.target
                XCTAssertTrue((kind == .trail ? 5...100 : 1...10).contains(n))
                XCTAssertEqual(Set(play.lesson.choices).count, play.lesson.choices.count)
                XCTAssertTrue(play.lesson.choices.contains(n))
                play.advance(); XCTAssertEqual(play.round, index)
                switch kind {
                case .share, .trail, .more, .tickets:
                    for wrong in play.lesson.choices where wrong != n {
                        play.choose(wrong); record(play); XCTAssertFalse(play.solved)
                    }
                    if kind == .more { XCTAssertEqual(n, play.lesson.fewer ? play.lesson.choices.min() : play.lesson.choices.max()) }
                    play.choose(n); record(play)
                }
                XCTAssertTrue(play.solved, "\(kind) level \(index)")
                let before = play.feedbackRevision
                play.choose(n)
                XCTAssertEqual(play.feedbackRevision, before)
                play.advance()
            }
            XCTAssertTrue(play.complete)
            play.advance(); XCTAssertEqual(play.round, play.total - 1)
            play.replay(); XCTAssertEqual(play.round, 0); XCTAssertFalse(play.complete); XCTAssertFalse(play.solved)
        }
        for text in narration {
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)
            XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: text)?.voice, "Ruth", text)
        }
    }

    @MainActor
    func testRemainingCountingDifficultyAndReplayVariety() {
        for kind in CountingActivity.allCases {
            for _ in 0..<20 {
                let levels = kind.levels()
                if ![.share, .more, .trail].contains(kind) {
                    XCTAssertEqual(levels.map(\.target), levels.map(\.target).sorted())
                }
                if kind == .trail {
                    XCTAssertEqual(Set(levels.map(\.goal)), Set(stride(from: 5, through: 100, by: 5)))
                    XCTAssertTrue(levels.contains { $0.start == 0 })
                    for lesson in levels {
                        XCTAssertLessThan(lesson.start, lesson.goal)
                        XCTAssertTrue(lesson.start.isMultiple(of: 5))
                        XCTAssertEqual(lesson.target, lesson.goal - lesson.start)
                        XCTAssertEqual(Set(lesson.choices).count, 3)
                        XCTAssertTrue(lesson.choices.contains(lesson.target))
                        XCTAssertTrue(lesson.choices.allSatisfy { $0.isMultiple(of: 5) })
                    }
                }
                if kind == .share {
                    XCTAssertEqual(levels.count, 10)
                    XCTAssertEqual(levels.filter(\.picnicAdds).count, 5)
                    XCTAssertEqual(Set(levels.map { "\($0.start):\($0.goal)" }).count, 10)
                    XCTAssertEqual(Set(levels.map { $0.food.id }).count, 10)
                    for lesson in levels {
                        XCTAssertEqual(lesson.target, abs(lesson.goal - lesson.start))
                        XCTAssertEqual(lesson.choices.count, 3)
                        XCTAssertTrue((1...10).contains(lesson.start))
                        XCTAssertTrue((1...10).contains(lesson.goal))
                        XCTAssertTrue(lesson.choices.allSatisfy { (0...10).contains(lesson.picnicAdds ? lesson.start + $0 : lesson.start - $0) })
                    }
                }
                if kind == .more {
                    XCTAssertTrue(levels.prefix(10).allSatisfy { !$0.fewer })
                    XCTAssertTrue(levels.suffix(10).allSatisfy { $0.fewer })
                    XCTAssertTrue(levels.allSatisfy { $0.target != $0.other })
                }
                for level in levels { XCTAssertTrue(level.choices.allSatisfy { (kind == .trail ? 0...100 : 1...12).contains($0) }) }
            }
        }
    }

            previous = play.rounds[0].signature
            for index in 0..<18 {
                let notes = play.current.notes
                XCTAssertEqual(notes.count, 2 + index / 3)
                XCTAssertGreaterThanOrEqual(Set(notes).count, 2)
                XCTAssertEqual(play.phase, .showing)
                XCTAssertEqual(play.tap(notes[0]), .ignored)
                play.next(); XCTAssertEqual(play.index, index)
                for wrongPosition in notes.indices {
                    play.finishDemo()
                    for note in notes.prefix(wrongPosition) { XCTAssertEqual(play.tap(note), .correct) }
                    let wrong = PatternDrum.allCases.first { $0 != notes[wrongPosition] }!
                    XCTAssertEqual(play.tap(wrong), .retry)
                    XCTAssertEqual(play.position, 0); XCTAssertEqual(play.phase, .showing)
                    XCTAssertEqual(play.tap(notes[0]), .ignored)
                    XCTAssertEqual(play.current.notes, notes)
                }
                play.finishDemo(); _ = play.tap(notes[0]); play.beginDemo()
                XCTAssertEqual(play.position, 0); XCTAssertEqual(play.phase, .showing)
                play.finishDemo()
                for (position, note) in notes.enumerated() {
                    XCTAssertEqual(play.tap(note), position == notes.count - 1 ? .solved : .correct)
                }
                XCTAssertEqual(play.phase, .solved)
                XCTAssertEqual(play.tap(notes[0]), .ignored)
                play.beginDemo(); play.finishDemo(); XCTAssertEqual(play.phase, .solved)
                play.next()
            }
            XCTAssertTrue(play.finished)
            play.next(); XCTAssertEqual(play.index, 18)
        }
        for drum in PatternDrum.allCases {
            let url = try XCTUnwrap(Bundle.main.url(forResource: drum.soundName, withExtension: "wav"))
            let audio = try AVAudioPlayer(contentsOf: url)
            XCTAssertTrue(audio.prepareToPlay()); XCTAssertEqual(audio.duration, 0.5, accuracy: 0.01)
        }
        for text in [DrumPatternRound.prompt, DrumPatternRound.watch, DrumPatternRound.ready, DrumPatternRound.retry, DrumPatternRound.success, DrumPatternRound.completion] + PatternDrum.allCases.map(\.name) {
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)

        }
    }

            XCTAssertNotEqual(play.rounds[0].signature, previous)
            previous = play.rounds[0].signature
            for index in 0..<18 {
                let route = play.current
                XCTAssertEqual(route.size, 4 + index / 6)
                XCTAssertEqual(route.path.count, route.size)
                XCTAssertEqual(route.vehicles.count, route.size * route.size)
                XCTAssertTrue(route.path.allSatisfy { (0..<route.size).contains($0) })
                XCTAssertEqual(play.row, 0)
                play.next(); XCTAssertEqual(play.index, index)
                for row in 1..<route.size {
                    XCTAssertLessThanOrEqual(abs(route.path[row] - route.path[row - 1]), 1)
                    for wrongRow in -1...route.size { for col in -1...route.size {
                        if wrongRow == row && col == route.path[row] { continue }
                        XCTAssertFalse(play.hop(row: wrongRow, column: col))
                        XCTAssertEqual(play.row, row - 1)
                    } }
                    XCTAssertTrue(play.hop(row: row, column: route.path[row]))
                    XCTAssertEqual(play.column, route.path[row])
                    XCTAssertFalse(play.hop(row: row, column: route.path[row]))
                }
                XCTAssertTrue(play.solved)
                XCTAssertFalse(play.hop(row: route.size, column: 0))
                play.next()
            }
            XCTAssertTrue(play.finished)
            play.next(); XCTAssertEqual(play.index, 18)
            XCTAssertFalse(play.hop(row: 0, column: 0))
        }
        for asset in FrogCrossing.vehicleAssets { XCTAssertNotNil(UIImage(named: asset), asset) }
        for text in [FrogCrossing.prompt, FrogCrossing.retry, FrogCrossing.blocked, FrogCrossing.success, FrogCrossing.completion] + (1...4).map(String.init) {
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)

        }
    }

            for index in 0..<24 {

                XCTAssertEqual(puzzle.size, 3 + index / 3)
                XCTAssertTrue((0...(puzzle.size - puzzle.patchSize)).contains(puzzle.row))
                XCTAssertTrue((0...(puzzle.size - puzzle.patchSize)).contains(puzzle.column))
                XCTAssertEqual(puzzle.choices.count, 3)

                }
            }

        }
        let old = Array(bank.prefix(21).map(\.word))

    @MainActor
    func testLetterDrawUsesEveryLetterOnceAndRestartsWithDifferentFirst() {
        for previous in [""] + LetterDrawAlphabet.letters {
            let letters = LetterDrawAlphabet.shuffled(startingAfter: previous, lastShown: "Z")
            XCTAssertEqual(letters.count, 26)
            XCTAssertEqual(Set(letters), Set(LetterDrawAlphabet.letters))
            XCTAssertNotEqual(letters.first, previous)
            XCTAssertNotEqual(letters.first, "Z")
            let play = LiteracyPlay(total: letters.count)
            var seen: Set<String> = []
            for index in 0..<26 {
                XCTAssertFalse(play.complete)
                XCTAssertEqual(play.round, index)
                XCTAssertTrue(seen.insert(letters[play.round]).inserted)
                play.advance()
                XCTAssertEqual(play.round, index, "Unfinished letters cannot be skipped")
                play.win("Done")
                play.advance()
            }
            XCTAssertTrue(play.complete)
            play.advance()
            XCTAssertEqual(play.round, 25)
            play.replay()
            XCTAssertEqual(play.round, 0)
            XCTAssertFalse(play.solved)
            XCTAssertFalse(play.complete)
        }
        XCTAssertEqual(LiteracyPlay().total, 3, "Other literacy games retain their round count")
    }
