    @MainActor
    func testCozyRoutinePathsOrderingActionsPauseAndRuth() throws {
        XCTAssertEqual(CozyRoutine.bank.count, 8); XCTAssertEqual(CozyRoutine.bank.reduce(0) { $0 + $1.steps.count }, 35)
        XCTAssertEqual(Set(CozyRoutineStep.bank.map(\.id)).count, 28)
        var layouts = Set<String>()
        for _ in 0..<30 {
            var play = CozyRoutinePlay()
            play.choosePath("invalid"); play.start(); play.next(); play.finish(); XCTAssertEqual(play.phase, .paths)
            for path in CozyRoutine.bank {
                XCTAssertEqual(Set(path.steps).count, path.steps.count)
                play.choosePath(path.id); XCTAssertEqual(play.phase, .introduction)
                XCTAssertEqual(play.prompt, path.introduction)
                play.start(); XCTAssertEqual(play.phase, .choosing)
                for (index, id) in path.steps.enumerated() {
                    XCTAssertEqual(play.current.id, id); XCTAssertEqual(play.index, index)
                    XCTAssertEqual(play.actions, 0); XCTAssertTrue(play.used.isEmpty)
                    XCTAssertEqual(play.choices.count, 3); XCTAssertEqual(Set(play.choices).count, 3)
                    XCTAssertEqual(Set(play.choices.map { CozyRoutineStep.named($0).symbol }).count, 3)
                    layouts.insert(play.choices.joined())
                    play.next(); XCTAssertEqual(play.index, index)
                    for wrong in play.choices where wrong != id { XCTAssertFalse(play.chooseStep(wrong)) }
                    XCTAssertTrue(play.chooseStep(id)); XCTAssertFalse(play.chooseStep(id))
                    XCTAssertEqual(play.phase, .acting)
                    let step = play.current
                    let order = [.pack, .scrub].contains(step.kind) ? Array((0..<step.goal).reversed()) : Array(0..<step.goal)
                    for value in order {
                        let before = play.actions, token = play.token
                        XCTAssertFalse(play.act(value, token: "stale")); XCTAssertFalse(play.act(-1, token: token))
                        if ![.pack, .scrub].contains(step.kind) { XCTAssertFalse(play.act(step.goal, token: token)) }
                        for used in play.used { XCTAssertFalse(play.act(used, token: token)) }
                        XCTAssertTrue(play.act(value, token: token)); XCTAssertEqual(play.actions, before + 1)
                        XCTAssertFalse(play.act(value, token: token)); XCTAssertEqual(play.actions, before + 1)
                    }
                    XCTAssertEqual(play.phase, .feedback); XCTAssertEqual(play.prompt, step.response)
                    play.next(); play.next()
                    XCTAssertEqual(play.phase, index == path.steps.count - 1 ? .finished : .choosing)
                }
                XCTAssertTrue(play.completed.contains(path.id)); XCTAssertEqual(play.prompt, path.outcome)
                play.paths(); XCTAssertEqual(play.phase, .paths); XCTAssertNil(play.routine)
            }
            XCTAssertEqual(play.completed.count, 8)
            play.choosePath("bathroom"); play.start(); play.chooseStep("tell"); play.pause()
            XCTAssertEqual(play.phase, .paused); play.paths(); XCTAssertEqual(play.completed.count, 8)
        }
        XCTAssertGreaterThan(layouts.count, 100)
        for mode in 0..<4 {
            var play = CozyRoutinePlay(); play.choosePath("bathroom")
            if mode > 0 { play.start() }
            if mode > 1 { play.chooseStep("tell") }
            if mode > 2 { play.act(0, token: play.token) }
            play.pause(); XCTAssertEqual(play.phase, .paused); XCTAssertTrue(play.completed.isEmpty)
            play.finish(); XCTAssertEqual(play.phase, .complete); XCTAssertTrue(play.completed.isEmpty)
            play.paths(); play.start(); play.next(); XCTAssertEqual(play.phase, .complete)
        }
        for step in CozyRoutineStep.bank {
            let renderer = ImageRenderer(content: CozyRoutineItemArt(step: step).padding(15).background(Color.cyan.opacity(0.1)))
            let shot = XCTAttachment(image: try XCTUnwrap(renderer.uiImage)); shot.name = "Routine item \(step.id)"; shot.lifetime = .keepAlways; add(shot)
            XCTAssertNotNil(UIImage(systemName: step.symbol), step.symbol)
            for symbol in step.items { XCTAssertNotNil(UIImage(systemName: symbol), symbol) }
        }
        for path in CozyRoutine.bank {
            for finished in [false, true] {
                let step = CozyRoutineStep.named(finished ? path.steps.last! : path.steps[0])
                let renderer = ImageRenderer(content: CozyRoutineScene(path: path, step: step, accomplished: finished, completedSteps: finished ? path.steps : []).frame(width: 350).background(.white))
                let shot = XCTAttachment(image: try XCTUnwrap(renderer.uiImage)); shot.name = "Routine \(path.id) \(finished ? "finished" : "start")"; shot.lifetime = .keepAlways; add(shot)
            }
        }
        for text in Set(CozyRoutine.narration) {
            let clip = try XCTUnwrap(NarrationAudioCatalog.shared.clip(for: text), text)
            XCTAssertEqual(clip.voice, "Ruth"); XCTAssertEqual(clip.engine, "generative")
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text))
        }
    }

    func testFriendshipBridgeAllActivitiesTurnGuardsReplayAndRuth() throws {

        XCTAssertEqual(Set(BridgeAdventure.bank.map(\.id)).count, 12)
        XCTAssertEqual(Set(BridgeAdventure.bank.map { $0.kind.rawValue }).count, 6)
        var previous = "", orders = Set<String>()
        for session in 0..<100 {
            var play = BridgePlay(previousFirst: previous)
            XCTAssertNotEqual(play.current.id, previous); previous = play.current.id
            orders.insert(play.adventures.map(\.id).joined())
            XCTAssertEqual(play.phase, .friends); XCTAssertEqual(play.bridgePieces, 0)
            play.start(); play.next(); play.chooseFriend("invalid"); XCTAssertEqual(play.phase, .friends)
            play.chooseFriend(TogetherFriend.bank[session % TogetherFriend.bank.count].id)
            let chosen = play.friend?.id
            for round in 0..<12 {
                XCTAssertEqual(play.index, round); XCTAssertEqual(play.bridgePieces, round)
                XCTAssertEqual(play.phase, .choosing); XCTAssertEqual(play.actions, 0)
                play.chooseWords(-1); play.chooseWords(2); XCTAssertEqual(play.phase, .choosing)
                play.chooseWords(session % 2); XCTAssertEqual(play.phase, .ready)
                XCTAssertEqual(play.prompt, play.current.choices[session % 2].response)
                play.next(); XCTAssertEqual(play.index, round)
                play.chooseFriend("maya"); XCTAssertEqual(play.friend?.id, chosen)
                play.start(); XCTAssertEqual(play.phase, .playing)
                XCTAssertEqual(Set(play.bubbles), Set(0..<6)); XCTAssertEqual(Set(play.copyChoices), [0, 1, 2])
                for i in 1..<4 { XCTAssertNotEqual(play.copySequence[i], play.copySequence[i - 1]) }
                for step in 0..<play.current.goal {
                    let token = play.token, turn = play.turn, kind = play.current.kind
                    XCTAssertFalse(play.act(-1, token: token)); XCTAssertEqual(play.actions, step)
                    XCTAssertFalse(play.act(0, token: "stale")); XCTAssertEqual(play.actions, step)
                    var value = 0, target = 0
                    switch kind {
                    case .wave: value = 0
                    case .pass:
                        XCTAssertFalse(play.act(1 - turn, token: token)); value = turn
                    case .copy:
                        XCTAssertFalse(play.act((play.copyTarget + 1) % 3, token: token)); value = play.copyTarget
                    case .build:
                        XCTAssertFalse(play.act(step, token: token, target: 1 - turn)); value = step; target = turn
                    case .tidy:
                        XCTAssertFalse(play.act(step, token: token, target: 1)); value = step
                    case .bubbles:
                        value = play.bubbles.first { $0 % 2 == turn && !play.popped.contains($0) }!
                        for popped in play.popped { XCTAssertFalse(play.act(popped, token: token)) }
                    }
                    XCTAssertTrue(play.act(value, token: token, target: target)); XCTAssertEqual(play.actions, step + 1)
                    XCTAssertFalse(play.act(value, token: token, target: target)); XCTAssertEqual(play.actions, step + 1)
                }
                XCTAssertEqual(play.phase, .celebration); XCTAssertEqual(play.bridgePieces, round + 1)
                XCTAssertEqual(play.prompt, play.current.outcome)
                play.start(); XCTAssertEqual(play.phase, .celebration)
                play.next(); play.next(); XCTAssertEqual(play.index, round + 1)
            }
            XCTAssertEqual(play.phase, .complete); XCTAssertEqual(play.bridgePieces, 12)
            play.chooseWords(0); play.start(); play.next(); XCTAssertEqual(play.phase, .complete)
        }
        XCTAssertEqual(orders.count, 100)
        for text in Set(BridgeAdventure.narration) {
            let clip = try XCTUnwrap(NarrationAudioCatalog.shared.clip(for: text), text)
            XCTAssertEqual(clip.voice, "Ruth"); XCTAssertEqual(clip.engine, "generative")
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text))
        }
        for friend in TogetherFriend.bank {
            for pieces in [0, 6, 12] {
                let renderer = ImageRenderer(content: BridgeFriendScene(friend: friend, pieces: pieces, waving: true).frame(width: 350).background(.white))
                let shot = XCTAttachment(image: try XCTUnwrap(renderer.uiImage)); shot.name = "Friendship \(friend.name) bridge \(pieces)"; shot.lifetime = .keepAlways; add(shot)
            }
        }
    }
