    @MainActor
    func testLaundryColorsCyclesAndRuth() {
        XCTAssertEqual(LaundryColor.allCases.count, 11)
        var session = LaundrySession()
        var previous: LaundryColor?
        for cycle in 0..<100 {
            var seen: Set<LaundryColor> = []
            for index in 0..<11 {
                let round = session.current
                if index == 0 { XCTAssertNotEqual(round.target, previous) }
                XCTAssertTrue(seen.insert(round.target).inserted)
                XCTAssertEqual(Set(round.baskets).count, 3)
                XCTAssertTrue(round.baskets.contains(round.target))
                session.next()
                XCTAssertEqual(session.current.target, round.target)
                for wrong in round.baskets where wrong != round.target {
                    XCTAssertFalse(session.sort(into: wrong))
                }
                XCTAssertEqual(session.total, cycle * 11 + index)
                XCTAssertTrue(session.sort(into: round.target))
                XCTAssertFalse(session.sort(into: round.target))
                XCTAssertEqual(session.total, cycle * 11 + index + 1)
                previous = round.target
                session.next()
            }
            XCTAssertEqual(seen, Set(LaundryColor.allCases))
        }
        for color in LaundryColor.allCases {
            for text in [color.prompt, color.success, color.retry] {
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)
                XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: text)?.voice, "Ruth")
            }
        }
    }

            }
        }
        var recent: [String] = []
        for _ in 0..<2000 {

            XCTAssertFalse(recent.contains(next.word))

            XCTAssertEqual(recent.last, next.word)
            // Match persisted history restored on the next game entry.
            recent = recent.joined(separator: "|").split(separator: "|").map(String.init)
        }
        let old = Array(bank.prefix(21).map(\.word))
        XCTAssertEqual(WordBuilderWord.remembering("CAT", recent: old), Array((old + ["CAT"]).suffix(20)))
    }

    @MainActor
    func testLetterDrawUsesEveryLetterOnceAndRestartsWithDifferentFirst() {
        for previous in [""] + LetterDrawAlphabet.letters {
            let letters = LetterDrawAlphabet.shuffled(startingAfter: previous, lastShown: "Z")
            XCTAssertEqual(letters.count, 26)
            XCTAssertEqual(Set(letters), Set(LetterDrawAlphabet.letters))
            XCTAssertNotEqual(letters.first, previous)
            XCTAssertNotEqual(letters.first, "Z")
            let play = LiteracyPlay(total: letters.count)
            var seen: Set<String> = []
            for index in 0..<26 {
                XCTAssertFalse(play.complete)
                XCTAssertEqual(play.round, index)
                XCTAssertTrue(seen.insert(letters[play.round]).inserted)
                play.advance()
                XCTAssertEqual(play.round, index, "Unfinished letters cannot be skipped")
                play.win("Done")
                play.advance()
            }
            XCTAssertTrue(play.complete)
            play.advance()
            XCTAssertEqual(play.round, 25)
            play.replay()
            XCTAssertEqual(play.round, 0)
            XCTAssertFalse(play.solved)
            XCTAssertFalse(play.complete)
        }
        XCTAssertEqual(LiteracyPlay().total, 3, "Other literacy games retain their round count")
    }
