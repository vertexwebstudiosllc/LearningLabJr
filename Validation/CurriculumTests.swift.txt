    func testShapeTrailsGeometryAndProgression() {
        XCTAssertEqual(ShapeTrail.bank.count, 25)
        XCTAssertEqual(Set(ShapeTrail.bank.map(\.id)).count, 25)
        for trail in ShapeTrail.bank {
            XCTAssertEqual(trail.points.first, trail.points.last, trail.name)
            XCTAssertEqual(trail.checkpoints.first, 0)
            XCTAssertEqual(trail.checkpoints.last, trail.points.count - 1)
            XCTAssertEqual(trail.checkpoints, trail.checkpoints.sorted())
            XCTAssertEqual(Set(trail.checkpoints).count, trail.checkpoints.count)
            XCTAssertTrue((5...40).contains(trail.checkpoints.count), trail.name)
            for point in trail.points {
                XCTAssertTrue((0.05...0.95).contains(point.x), trail.name)
                XCTAssertTrue((0.05...0.95).contains(point.y), trail.name)
            }
            for (a, b) in zip(trail.points, trail.points.dropFirst()) {
                XCTAssertGreaterThan(hypot(b.x-a.x, b.y-a.y), 0.00001, trail.name)
            }
            for size in [CGSize(width: 280, height: 280), CGSize(width: 362, height: 280), CGSize(width: 740, height: 280)] {
                let guide = trail.path(in: size)
                let complete = trail.path(in: size, visited: trail.checkpoints.count)
                XCTAssertEqual(guide.boundingRect, complete.boundingRect, trail.name)
                for stop in trail.checkpoints {
                    let point = trail.location(trail.points[stop], in: size)
                    XCTAssertTrue(CGRect(origin: .zero, size: size).contains(point), trail.name)
                }
            }
        }
        var session = TrailSession()
        let size = CGSize(width: 362, height: 280)
        for index in 0..<25 {
            session.next()
            XCTAssertEqual(session.index, index)
            let count = session.current.checkpoints.count
            for stop in 0..<count {
                XCTAssertFalse(session.touch(CGPoint(x: -100, y: -100), in: size))
                XCTAssertEqual(session.visited, stop)
                let point = session.current.location(session.current.points[session.current.checkpoints[stop]], in: size)
                XCTAssertTrue(session.touch(point, in: size))
                XCTAssertFalse(session.advance(expected: stop))
            }
            XCTAssertTrue(session.solved)
            session.next()
        }
        XCTAssertTrue(session.finished)
        XCTAssertFalse(session.advance(expected: session.visited))
        session.next()
        XCTAssertEqual(session.index, 25)
    }

    func testNestingColorSequencesAndShapeCycles() throws {
        var nest = NestSession()
        var previous: PostShape?
        var previousColors: [String: [String]] = [:]
        for _ in 0..<100 {
            var seen: Set<PostShape> = []
            for index in 0..<10 {
                let round = nest.current
                if index == 0 { XCTAssertNotEqual(round.shape, previous) }
                XCTAssertTrue(seen.insert(round.shape).inserted)
                XCTAssertEqual(Set(round.colors).count, 3)
                XCTAssertEqual(Set(round.choices), Set(round.colors))
                XCTAssertTrue(Set(round.colors).isSubset(of: Set(NestRound.palette)))
                XCTAssertNotEqual(round.colors, previousColors[round.shape.rawValue])
                nest.next()
                XCTAssertEqual(nest.current.shape, round.shape)
                for layer in 0..<3 {
                    for wrong in NestRound.palette where wrong != round.colors[layer] {
                        XCTAssertFalse(nest.choose(wrong, expectedLayer: layer))
                        XCTAssertEqual(nest.layer, layer)
                    }
                    XCTAssertTrue(nest.choose(round.colors[layer], expectedLayer: layer))
                    XCTAssertFalse(nest.choose(round.colors[layer], expectedLayer: layer))
                }
                XCTAssertTrue(nest.solved)
                previous = round.shape
                previousColors[round.shape.rawValue] = round.colors
                nest.next()
            }
            XCTAssertEqual(seen, Set(PostShape.allCases))
        }
        let data = try JSONEncoder().encode(nest.memory)
        let memory = try JSONDecoder().decode(NestMemory.self, from: data)
        let restarted = NestSession(memory: memory)
        XCTAssertNotEqual(restarted.current.shape.rawValue, memory.lastShape)
        XCTAssertNotEqual(restarted.current.colors, memory.palettes[restarted.current.shape.rawValue])
        XCTAssertEqual(restarted.layer, 0)
        XCTAssertEqual(restarted.completed, 0)
    }

    func testShapeTownPicturesAndAnyOrderDrops() {
        XCTAssertEqual(TownPicture.bank.count, 20)
        XCTAssertEqual(Set(TownPicture.bank.map(\.id)).count, 20)
        XCTAssertEqual(Set(TownPicture.bank.map { $0.pieces.count }), [3, 4, 5])
        for width: CGFloat in [280, 362, 740] {
            for reverse in [false, true] {
                var town = TownSession()
                for picture in TownPicture.bank {
                    XCTAssertEqual(town.picture.id, picture.id)
                    XCTAssertFalse(town.completePicture)
                    town.next()
                    XCTAssertEqual(town.picture.id, picture.id)
                    let pieces = reverse ? Array(picture.pieces.reversed()) : picture.pieces
                    for piece in pieces {
                        let frame = piece.frame(boardWidth: width)
                        XCTAssertGreaterThanOrEqual(frame.minX, 0)
                        XCTAssertLessThanOrEqual(frame.maxX, width)
                        XCTAssertGreaterThanOrEqual(frame.minY, 0)
                        XCTAssertLessThanOrEqual(frame.maxY, 230)
                        XCTAssertTrue(LabPaint.bank.contains { $0.id == piece.paint })
                        if piece.shape == .square || piece.shape == .circle { XCTAssertEqual(piece.width, piece.height) }
                        XCTAssertFalse(town.place(pieceID: piece.id, at: CGPoint(x: width / 2, y: 275), boardWidth: width))
                        XCTAssertFalse(town.place(pieceID: 999, at: CGPoint(x: frame.midX, y: frame.midY), boardWidth: width))
                        XCTAssertTrue(town.place(pieceID: piece.id, at: CGPoint(x: frame.midX, y: frame.midY), boardWidth: width))
                        XCTAssertFalse(town.place(pieceID: piece.id, at: CGPoint(x: frame.midX, y: frame.midY), boardWidth: width))
                    }
                    XCTAssertTrue(town.completePicture)
                    town.next()
                }
                XCTAssertTrue(town.finished)
                town.next()
                XCTAssertEqual(town.index, 20)
            }
        }
        var town = TownSession()
        // A circle window cannot be dropped into the triangular roof.
        let roof = town.picture.pieces[1].frame(boardWidth: 320)
        XCTAssertFalse(town.place(pieceID: 2, at: CGPoint(x: roof.midX, y: roof.midY), boardWidth: 320))
        for picture in TownPicture.bank {
            if picture.id == "car" {
                let rightWheel = picture.pieces[3].frame(boardWidth: 320)
                XCTAssertTrue(town.place(pieceID: 2, at: CGPoint(x: rightWheel.midX, y: rightWheel.midY), boardWidth: 320))
                XCTAssertTrue(town.placed.contains(3))
                let leftWheel = picture.pieces[2].frame(boardWidth: 320)
                XCTAssertTrue(town.place(pieceID: 3, at: CGPoint(x: leftWheel.midX, y: leftWheel.midY), boardWidth: 320))
                break
            }
            for piece in picture.pieces {
                let frame = piece.frame(boardWidth: 320)
                town.place(pieceID: piece.id, at: CGPoint(x: frame.midX, y: frame.midY), boardWidth: 320)
            }
            town.next()
        }
    }

    func testPatternTrainProgressionAndUniqueSequences() {
        for _ in 0..<100 {
            let rounds = TrainPattern.session()
            XCTAssertEqual(rounds.count, 60)
            XCTAssertEqual(Set(rounds.map(\.key)).count, 60)
            XCTAssertEqual(Set(rounds.map { $0.shapes.map(\.rawValue).joined(separator: ",") }).count, 60)
            XCTAssertEqual(Set(rounds.map(\.count)), Set(5...15))
            var previousCount = 0
            for (index, round) in rounds.enumerated() {
                XCTAssertEqual(round.stage, index / 12)
                XCTAssertGreaterThanOrEqual(round.count, previousCount)
                previousCount = round.count
                XCTAssertGreaterThanOrEqual(round.count - 1, round.unit.count * 2)
                XCTAssertEqual(round.answer, round.unit[(round.count - 1) % round.unit.count])
                XCTAssertEqual(Set(round.choices).count, 3)
                XCTAssertTrue(round.choices.contains(round.answer))
                for i in round.shapes.indices { XCTAssertEqual(round.shapes[i], round.unit[i % round.unit.count]) }
            }
        }
        var train = TrainSession()
        for index in 0..<60 {
            train.next()
            XCTAssertEqual(train.index, index)
            for wrong in TrainShape.allCases where wrong != train.current.answer { XCTAssertFalse(train.choose(wrong)) }
            XCTAssertFalse(train.solved)
            XCTAssertTrue(train.choose(train.current.answer))
            XCTAssertFalse(train.choose(train.current.answer))
            train.next()
        }
        XCTAssertTrue(train.finished)
        XCTAssertFalse(train.choose(train.current.answer))
        train.next()
        XCTAssertEqual(train.index, 60)
    }

    func testColorLabLevelsAndMixing() {
        XCTAssertEqual(LabLevel.levels.count, 9)
        XCTAssertEqual(Set(LabPaint.bank.map(\.id)).count, LabPaint.bank.count)
        var lab = ColorLabSession()
        for (index, level) in LabLevel.levels.enumerated() {
            XCTAssertEqual(lab.levelIndex, index)
            XCTAssertEqual(Set(level.paints).count, 3)
            XCTAssertEqual(Set(level.recipes.map(\.output)).count, 3)
            XCTAssertNil(lab.mix())
            lab.next()
            XCTAssertEqual(lab.levelIndex, index)
            lab.select("invalid")
            XCTAssertTrue(lab.selected.isEmpty)
            lab.select(level.paints[0]); lab.select(level.paints[0])
            XCTAssertTrue(lab.selected.isEmpty)
            for id in level.paints { lab.select(id) }
            XCTAssertEqual(lab.selected, Array(level.paints.suffix(2)))
            for id in lab.selected { lab.select(id) }
            for (number, recipe) in level.recipes.enumerated() {
                XCTAssertTrue(recipe.ingredients.isSubset(of: Set(level.paints)))
                for order in [[recipe.first, recipe.second], [recipe.second, recipe.first]] {
                    if lab.levelComplete { break }
                    for id in order { lab.select(id) }
                    XCTAssertEqual(lab.mix()?.output, recipe.output)
                    XCTAssertEqual(lab.result?.id, recipe.output)
                    XCTAssertEqual(lab.discoveries.count, number + 1)
                    XCTAssertTrue(lab.selected.isEmpty)
                }
            }
            XCTAssertTrue(lab.levelComplete)
            lab.next()
            if !lab.finished {
                XCTAssertTrue(lab.discoveries.isEmpty)
                XCTAssertNil(lab.result)
            }
        }
        XCTAssertTrue(lab.finished)
        XCTAssertNil(lab.mix())
        lab.next()
        XCTAssertEqual(lab.levelIndex, 9)
        let blue = LabLevel.levels.first { $0.title == "Blue shades" }!
        XCTAssertEqual(Set(blue.recipes.map(\.output)), ["navy", "gray", "sky-blue"])
    }

        XCTAssertEqual(LaundryColor.allCases.count, 11)
        var session = LaundrySession()
        var previous: LaundryColor?
        for cycle in 0..<100 {
            var seen: Set<LaundryColor> = []
            for index in 0..<11 {
                let round = session.current
                if index == 0 { XCTAssertNotEqual(round.target, previous) }
                XCTAssertTrue(seen.insert(round.target).inserted)
                XCTAssertEqual(Set(round.baskets).count, 3)
                XCTAssertTrue(round.baskets.contains(round.target))
                for width: CGFloat in [280, 362, 740] {
                    for (position, color) in round.baskets.enumerated() {
                        let frame = LaundryRound.basketFrame(index: position, width: width)
                        XCTAssertGreaterThanOrEqual(frame.width, 64)
                        XCTAssertEqual(round.basket(at: CGPoint(x: frame.midX, y: frame.midY), width: width), color)
                    }
                    XCTAssertNil(round.basket(at: CGPoint(x: width / 2, y: 70), width: width))
                    XCTAssertNil(round.basket(at: CGPoint(x: -10, y: 200), width: width))
                }
                session.next()
                XCTAssertEqual(session.current.target, round.target)
                for wrong in round.baskets where wrong != round.target {
                    XCTAssertFalse(session.sort(into: wrong))
                }
                XCTAssertEqual(session.total, cycle * 11 + index)
                XCTAssertTrue(session.sort(into: round.target))
                XCTAssertFalse(session.sort(into: round.target))
                XCTAssertEqual(session.total, cycle * 11 + index + 1)
                previous = round.target
                session.next()
            }
            XCTAssertEqual(seen, Set(LaundryColor.allCases))
        }
        for color in LaundryColor.allCases {
            for text in [color.prompt, color.success, color.retry] {
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: text), text)
                XCTAssertEqual(NarrationAudioCatalog.shared.clip(for: text)?.voice, "Ruth")
            }
        }
    }

        XCTAssertEqual(Set(bank.map { $0.letters.count }), [3, 4, 5])

        for item in bank {
            XCTAssertNotNil(UIImage(named: item.asset) ?? UIImage(named: "Space/\(item.asset)"), item.asset)
            XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: item.prompt), item.prompt)

            for index in item.letters.indices {
                let letter = item.letters[index]
                XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: "Look at the model. Find \(letter) next."))
                if index + 1 < item.letters.count {
                    XCTAssertNotNil(NarrationAudioCatalog.shared.url(for: "\(letter). Next find \(item.letters[index + 1])."))
                }
            }
        }
        var recent: [String] = []
        for _ in 0..<2000 {
            let next = WordBuilderWord.next(recent: recent)
            XCTAssertFalse(recent.contains(next.word))
            recent = WordBuilderWord.remembering(next.word, recent: recent)
            XCTAssertLessThanOrEqual(recent.count, 20)
            XCTAssertEqual(recent.last, next.word)
            // Match persisted history restored on the next game entry.
            recent = recent.joined(separator: "|").split(separator: "|").map(String.init)
        }
        let old = Array(bank.prefix(21).map(\.word))
        XCTAssertEqual(WordBuilderWord.remembering("CAT", recent: old), Array((old + ["CAT"]).suffix(20)))
    }

    @MainActor
    func testLetterDrawUsesEveryLetterOnceAndRestartsWithDifferentFirst() {
        for previous in [""] + LetterDrawAlphabet.letters {
            let letters = LetterDrawAlphabet.shuffled(startingAfter: previous, lastShown: "Z")
            XCTAssertEqual(letters.count, 26)
            XCTAssertEqual(Set(letters), Set(LetterDrawAlphabet.letters))
            XCTAssertNotEqual(letters.first, previous)
            XCTAssertNotEqual(letters.first, "Z")
            let play = LiteracyPlay(total: letters.count)
            var seen: Set<String> = []
            for index in 0..<26 {
                XCTAssertFalse(play.complete)
                XCTAssertEqual(play.round, index)
                XCTAssertTrue(seen.insert(letters[play.round]).inserted)
                play.advance()
                XCTAssertEqual(play.round, index, "Unfinished letters cannot be skipped")
                play.win("Done")
                play.advance()
            }
            XCTAssertTrue(play.complete)
            play.advance()
            XCTAssertEqual(play.round, 25)
            play.replay()
            XCTAssertEqual(play.round, 0)
            XCTAssertFalse(play.solved)
            XCTAssertFalse(play.complete)
        }
        XCTAssertEqual(LiteracyPlay().total, 3, "Other literacy games retain their round count")
    }
