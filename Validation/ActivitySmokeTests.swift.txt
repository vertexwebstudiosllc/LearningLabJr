    @MainActor
    func testCozyRoutinesEightPathsDragWashingPauseAndReplay() async throws {
        let app = try await launch()
        func tap(_ button: XCUIElement) throws {
            try reveal(button, in: app)
            if button.frame.maxY > app.windows.firstMatch.frame.maxY - 40 { app.scrollViews.firstMatch.swipeUp() }
            try reveal(button, in: app); button.tap()
        }
        let paths: [String: [String]] = [
            "bedtime": ["wash", "teeth", "pajamas", "book", "bed"],
            "bathroom": ["tell", "potty", "wipe", "clothes", "flush", "wash"],
            "rain": ["raincoat", "boots", "umbrella", "walk"],
            "snow": ["warmcoat", "hat", "snowboots", "walk"],
            "beach": ["suncream", "sunhat", "beachbag", "shade"],
            "morning": ["clothes", "breakfast", "teeth", "hair"],
            "picnic": ["wash", "picnicbag", "walk", "cleanup"],
            "home": ["shoesoff", "coatoff", "wash", "toys"]
        ]
        let kinds = ["wash": "wash", "teeth": "scrub", "pajamas": "dress", "book": "pages", "bed": "tap", "tell": "tap", "potty": "tap", "wipe": "tap", "clothes": "dress", "flush": "tap", "raincoat": "dress", "boots": "dress", "umbrella": "dress", "walk": "walk", "warmcoat": "dress", "hat": "dress", "snowboots": "dress", "suncream": "scrub", "sunhat": "dress", "beachbag": "pack", "shade": "tap", "breakfast": "tap", "hair": "scrub", "picnicbag": "pack", "cleanup": "pack", "shoesoff": "dress", "coatoff": "dress", "toys": "pack"]
        try tap(app.buttons["category.Big Feelings"])
        let card = app.buttons["activity-card.Cozy Evening Path"]
        try tap(card)
        try tap(app.buttons["routine.path.bathroom"]); try tap(app.buttons["routine.start"])
        try tap(app.buttons["routine.pause"]); try tap(app.buttons["routine.paths"])
        XCTAssertEqual(app.staticTexts["routine.collection"].label, "0 of 8 paths explored")
        var completed = 0, dragged = false
        for path in ["bedtime", "bathroom", "rain", "snow", "beach", "morning", "picnic", "home"] {
            try tap(app.buttons["routine.path.\(path)"]); try tap(app.buttons["routine.start"])
            let steps = paths[path]!
            for (index, step) in steps.enumerated() {
                XCTAssertEqual(app.staticTexts["routine.progress"].label, "Step \(index + 1) of \(steps.count)")
                if index == 0 {
                    let wrong = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'routine.choice.'")).allElementsBoundByIndex.first { $0.identifier != "routine.choice.\(step)" }!
                    try tap(wrong); XCTAssertTrue(app.staticTexts["routine.feedback"].exists)
                }
                try tap(app.buttons["routine.choice.\(step)"])
                let kind = kinds[step]!
                let goal = kind == "wash" ? 5 : ["scrub", "pages", "walk", "pack"].contains(kind) ? 3 : 1
                let order = ["pack", "scrub"].contains(kind) ? Array((0..<goal).reversed()) : Array(0..<goal)
                for value in order {
                    if ["dress", "pack"].contains(kind) {
                        let item = app.buttons["routine.item.\(value)"], drop = app.buttons["routine.drop"]
                        if !dragged {
                            try reveal(drop, in: app); try reveal(item, in: app)
                            item.press(forDuration: 0.6, thenDragTo: drop); dragged = true
                        } else { try tap(item); try tap(drop) }
                    } else {
                        if kind == "wash" && value == 0 {
                            try tap(app.buttons["routine.action.4"])
                            XCTAssertEqual(app.staticTexts["routine.actions"].label, "0 of 5 actions")
                        }
                        try tap(app.buttons["routine.action.\(value)"])
                    }
                }
                XCTAssertTrue(app.buttons["routine.next"].waitForExistence(timeout: 5))
                if index == steps.count - 1 {
                    let shot = XCTAttachment(screenshot: app.screenshot()); shot.name = "Cozy path \(path) last step"; shot.lifetime = .keepAlways; add(shot)
                }
                try tap(app.buttons["routine.next"])
            }
            completed += 1
            try tap(app.buttons["routine.paths"])
            XCTAssertEqual(app.staticTexts["routine.collection"].label, "\(completed) of 8 paths explored")
        }
        XCTAssertTrue(dragged)
        try tap(app.buttons["routine.path.bathroom"]); try tap(app.buttons["routine.pause"])
        try tap(app.buttons["routine.finish"])
        XCTAssertTrue(app.staticTexts["We did it together!"].waitForExistence(timeout: 5))
        try tap(app.buttons["Play again"])
        XCTAssertEqual(app.staticTexts["routine.collection"].label, "0 of 8 paths explored")
        app.navigationBars.buttons.firstMatch.tap(); try tap(card)
        XCTAssertEqual(app.staticTexts["routine.collection"].label, "0 of 8 paths explored")
    }


    @MainActor
    func testFriendshipBridgeTwelvePlaydatesDragTurnsSillyGamesAndReplay() async throws {
        let app = try await launch()
        func tap(_ button: XCUIElement) throws {
            try reveal(button, in: app)
            if button.frame.maxY > app.windows.firstMatch.frame.maxY - 40 { app.scrollViews.firstMatch.swipeUp() }
            try reveal(button, in: app); button.tap()
        }
        let kinds = ["hello": "wave", "space": "wave", "tower": "build", "bridge": "build", "ball": "pass", "car": "pass", "faces": "copy", "animals": "copy", "bubbles": "bubbles", "stars": "bubbles", "tidy": "tidy", "picture": "tidy"]
        try tap(app.buttons["category.Big Feelings"])
        let card = app.buttons["activity-card.Friendship Bridge"]
        try tap(card); try tap(app.buttons["bridge.friend.kai"])
        var seen = Set<String>(), dragged = false
        for round in 0..<12 {
            let story = app.staticTexts.matching(NSPredicate(format: "identifier BEGINSWITH 'bridge.story.'")).firstMatch
            XCTAssertTrue(story.waitForExistence(timeout: 5))
            let id = String(story.identifier.dropFirst("bridge.story.".count))
            XCTAssertTrue(seen.insert(id).inserted)
            let kind = try XCTUnwrap(kinds[id])
            XCTAssertEqual(app.staticTexts["bridge.progress"].label, "\(round) of 12 bridge pieces")
            try tap(app.buttons["bridge.words.\(round % 2)"])
            try tap(app.buttons["bridge.start"])
            let goal = kind == "wave" ? 3 : ["copy", "tidy"].contains(kind) ? 4 : 6
            var popped = Set<Int>()
            for step in 0..<goal {
                switch kind {
                case "wave": try tap(app.buttons["bridge.wave"])
                case "pass":
                    let toy = app.buttons["bridge.pass"]
                    try reveal(toy, in: app)
                    let beforeY = toy.frame.midY
                    toy.tap()
                    if step < goal - 1 { XCTAssertEqual(toy.frame.midY, beforeY, accuracy: 45, "Passing should keep the play area in view") }
                case "copy":
                    let example = app.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH 'bridge.example.'")).firstMatch
                    let answer = try XCTUnwrap(Int(example.identifier.split(separator: ".").last!))
                    if step == 0 {
                        try tap(app.buttons["bridge.copy.\((answer + 1) % 3)"])
                        XCTAssertEqual(app.staticTexts["bridge.actions"].label, "0 of 4 playful actions")
                    }
                    try tap(app.buttons["bridge.copy.\(answer)"])
                case "bubbles":
                    if step == 0 {
                        try tap(app.buttons["bridge.bubble.1"])
                        XCTAssertEqual(app.staticTexts["bridge.actions"].label, "0 of 6 playful actions")
                    }
                    let bubble = (0..<6).first { $0 % 2 == step % 2 && !popped.contains($0) }!
                    try tap(app.buttons["bridge.bubble.\(bubble)"]); popped.insert(bubble)
                default:
                    let target = kind == "build" ? step % 2 : 0
                    let source = app.buttons["bridge.drag"], destination = app.buttons["bridge.drop.\(target)"]
                    if !dragged {
                        try reveal(destination, in: app); try reveal(source, in: app)
                        source.press(forDuration: 0.6, thenDragTo: destination)
                        XCTAssertEqual(app.staticTexts["bridge.actions"].label, "1 of \(goal) playful actions")
                        dragged = true
                    } else {
                        try tap(source)
                        if kind == "build" && step == 0 {
                            try tap(app.buttons["bridge.drop.1"])
                            XCTAssertEqual(app.staticTexts["bridge.actions"].label, "0 of 6 playful actions")
                        }
                        try tap(destination)
                    }
                }
                if step == 1 {
                    let shot = XCTAttachment(screenshot: app.screenshot()); shot.name = "Friendship \(id) playing"; shot.lifetime = .keepAlways; add(shot)
                }
            }
            XCTAssertTrue(app.buttons["bridge.next"].waitForExistence(timeout: 5))
            XCTAssertEqual(app.staticTexts["bridge.progress"].label, "\(round + 1) of 12 bridge pieces")
            try tap(app.buttons["bridge.next"])
        }
        XCTAssertTrue(dragged); XCTAssertEqual(seen.count, 12)
        XCTAssertTrue(app.staticTexts["We did it together!"].waitForExistence(timeout: 5))
        try tap(app.buttons["Play again"])
        XCTAssertTrue(app.buttons["bridge.friend.maya"].exists)
        try tap(app.buttons["bridge.friend.maya"])
        XCTAssertEqual(app.staticTexts["bridge.progress"].label, "0 of 12 bridge pieces")
        app.navigationBars.buttons.firstMatch.tap(); try tap(card)
        XCTAssertTrue(app.buttons["bridge.friend.maya"].exists)
    }


    @MainActor
    func testBodyCluesAllPeopleDirectTapsHintsCompletionAndReplay() async throws {
        let app = try await launch()
        func tap(_ button: XCUIElement) throws {
            try reveal(button, in: app)
            if button.frame.maxY > app.windows.firstMatch.frame.maxY - 40 { app.scrollViews.firstMatch.swipeUp() }
            try reveal(button, in: app); button.tap()
        }
        let answers = ["hat": "head", "helmet": "head", "look": "eyes", "blink": "eyes", "listen": "ears", "earmuffs": "ears", "smell": "nose", "sniff": "nose", "bite": "mouth", "smile": "mouth", "wave": "hands", "mittens": "hands", "rumble": "tummy", "middle": "tummy", "bend": "knees", "kneepads": "knees", "socks": "feet", "toes": "feet"]

        try tap(card)

        for index in 0..<18 {
            let picture = app.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH 'body.picture.'")).firstMatch
            XCTAssertTrue(picture.waitForExistence(timeout: 5))
            let clue = String(picture.identifier.dropFirst("body.picture.".count))
            let buddy = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'body.buddy.'")).firstMatch
            XCTAssertTrue(seenPeople.insert(buddy.identifier).inserted)
            XCTAssertTrue(seenQuestions.insert(clue).inserted)
            XCTAssertEqual(app.staticTexts["body.progress"].label, "Clue \(index + 1) of 18")
            let part = try XCTUnwrap(answers[clue])
            if index == 0 {
                try tap(buddy)
                let wrong = part == "head" ? "tummy" : "head"
                try tap(app.buttons["body.part.\(wrong).0"])
                XCTAssertFalse(app.buttons["body.next"].exists)
                try tap(app.buttons["body.part.\(wrong).0"])
                XCTAssertTrue(app.staticTexts["body.feedback"].exists)
                let shot = XCTAttachment(screenshot: app.screenshot()); shot.name = "Body clue hint"; shot.lifetime = .keepAlways; add(shot)
            }
            if index == 1 { try tap(app.buttons["body.hint"]) }
            // Alternate sides to verify both members of paired body parts accept answers.
            let side = ["eyes", "ears", "hands", "knees", "feet"].contains(part) && index.isMultiple(of: 2) ? 1 : 0
            try tap(app.buttons["body.part.\(part).\(side)"])
            XCTAssertTrue(app.buttons["body.next"].waitForExistence(timeout: 5))
            XCTAssertEqual(app.staticTexts["body.answer"].label, part.capitalized)
            if index < 3 {
                let shot = XCTAttachment(screenshot: app.screenshot()); shot.name = "Body clue answer \(index)"; shot.lifetime = .keepAlways; add(shot)
            }
            try tap(app.buttons["body.next"])
        }
        XCTAssertEqual(seenPeople.count, 18); XCTAssertEqual(seenQuestions.count, 18)
        XCTAssertTrue(app.staticTexts["We did it together!"].waitForExistence(timeout: 5))
        try tap(app.buttons["Play again"])
        XCTAssertEqual(app.staticTexts["body.progress"].label, "Clue 1 of 18")
        let buddy = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'body.buddy.'")).firstMatch
        let prior = buddy.identifier
        app.navigationBars.buttons.firstMatch.tap(); try tap(card)
        XCTAssertNotEqual(buddy.identifier, prior)
        XCTAssertFalse(app.buttons["body.next"].exists)
    }

            "rain": ["book", "crayons"],
            "winter": ["coat", "hat"],

            "toys": ["box"],
            "plant": ["can", "water"],

            "puddle": ["umbrella", "boots"],
            "hungry": ["snack"],
            "thirsty": ["water", "cup"],

            let scene = app.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH 'solution.scene.'")).firstMatch

            let required = try XCTUnwrap(answers[id])
            if index == 0 {
                let wrong = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'solution.tool.'")).allElementsBoundByIndex.first { !required.contains(String($0.identifier.dropFirst("solution.tool.".count))) }!
                try tap(wrong)

                try tap(app.buttons["Close"])

            }
            for (offset, tool) in required.enumerated() {
                try tap(app.buttons["solution.tool.\(tool)"])
                if offset < required.count - 1 {
                    XCTAssertFalse(app.buttons["solution.tool.\(tool)"].isEnabled)
                    if index == 1 {
                        try tap(app.buttons["Close"]); try tap(app.buttons["solution.open"])
                        XCTAssertFalse(app.buttons["solution.tool.\(tool)"].isEnabled)
                    }

        app.navigationBars.buttons.firstMatch.tap(); try tap(card)
        XCTAssertNotEqual(scene.identifier, last)
        XCTAssertTrue(app.buttons["solution.open"].exists)
    }

    @MainActor
    private func reveal(_ element: XCUIElement, in app: XCUIApplication) throws {
        for _ in 0..<12 {
            let scroll = app.scrollViews.firstMatch
            let viewport = app.windows.firstMatch.frame
            let frame = element.exists ? element.frame : .null
            // XCTest can throw when asked about hittability above the visible window.
            if frame.intersects(viewport) && element.isHittable { return }
            let surface = scroll.exists ? scroll : app
            let movingDown = !frame.isNull && frame.midY < viewport.midY
            let start = surface.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: movingDown ? 0.4 : 0.7))
            let end = surface.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: movingDown ? 0.7 : 0.4))
            start.press(forDuration: 0.05, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.1)
        }
        let hierarchy = XCTAttachment(string: app.debugDescription)
        hierarchy.name = "Navigation failure hierarchy"
        hierarchy.lifetime = .keepAlways
        add(hierarchy)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Navigation failure screen"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        XCTFail("Could not reveal \(element)")
        throw NavigationFailure.missingControl(element.description)
    }

    @MainActor
    private func playExpandedCounting(_ kind: String, title: String, total: Int, previewLevels: Int? = nil) async throws {
        let app = try await launch()
        let category = app.buttons["category.123s & Counting"]
        try reveal(category, in: app); category.tap()
        let card = app.buttons["activity-card.\(title)"]
        try reveal(card, in: app); card.tap()
        func element(_ id: String) -> XCUIElement { app.descendants(matching: .any).matching(identifier: id).firstMatch }
        func number(_ label: String) -> Int { label.split(whereSeparator: { !$0.isNumber }).compactMap { Int($0) }.first! }
        func tap(_ id: String) throws { let button = app.buttons[id]; try reveal(button, in: app); button.tap() }
        var fruitPairs: Set<String> = []
        for index in 0..<(previewLevels ?? total) {
            let level = app.staticTexts["counting.ext.level"]
            XCTAssertTrue(level.waitForExistence(timeout: 5))
            XCTAssertEqual(level.label, "Level \(index + 1) of \(total)")
            let next = app.buttons["counting.ext.next"]
            XCTAssertFalse(next.exists)
            if previewLevels != nil {
                let shot = XCTAttachment(screenshot: app.screenshot())
                shot.name = title + " start \(index + 1)"; shot.lifetime = .keepAlways; add(shot)
            }
            switch kind {
            case "share":
                let have = number(app.staticTexts["counting.share.have"].label)
                let need = number(app.staticTexts["counting.share.need"].label)
                let target = abs(need - have)
                let choices = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'counting.ext.choice.'")).allElementsBoundByIndex
                XCTAssertEqual(choices.count, 3)
                XCTAssertEqual(Set(choices.map { number($0.label) }).count, 3)
                XCTAssertEqual(app.staticTexts["counting.ext.clue"].label, need > have ? "How many should we add?" : "How many should we take away?")
                try tap(choices.first { number($0.label) != target }!.identifier)
                XCTAssertEqual(level.label, "Level \(index + 1) of \(total)")
                XCTAssertTrue(app.staticTexts["counting.ext.feedback"].exists)
                if previewLevels != nil || index < 2 {
                    let shot = XCTAttachment(screenshot: app.screenshot())
                    shot.name = "Picnic math \(have) to \(need)"; shot.lifetime = .keepAlways; add(shot)
                }
                try tap("counting.ext.choice.\(target)")
                let advanced = XCTNSPredicateExpectation(predicate: NSPredicate(format: "label == %@", "Level \(index + 2) of \(total)"), object: level)
                if index + 1 < total { XCTAssertEqual(XCTWaiter.wait(for: [advanced], timeout: 8), .completed) }
                else { XCTAssertTrue(app.buttons["Play again"].waitForExistence(timeout: 8)) }
                continue
            case "trail":
                let left = element("counting.trail.left")
                let right = element("counting.trail.right")
                let start = number(left.label)
                let goal = number(right.label)
                let target = goal - start
                XCTAssertLessThan(start, goal)
                XCTAssertEqual(start % 5, 0); XCTAssertEqual(goal % 5, 0)
                let chests = app.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH 'counting.trail.chest.'")).allElementsBoundByIndex
                XCTAssertEqual(chests.count, 3)
                let correct = element("counting.trail.chest.\(target)")
                func drop(_ chest: XCUIElement, onto pile: XCUIElement) throws {
                    try reveal(chest, in: app)
                    chest.press(forDuration: 0.15, thenDragTo: pile)
                }
                try drop(correct, onto: right)
                XCTAssertFalse(next.exists); XCTAssertEqual(number(left.label), start)
                let wrong = chests.first { number($0.label) != target }!
                try drop(wrong, onto: left)
                XCTAssertFalse(next.exists); XCTAssertEqual(number(left.label), start)
                try drop(correct, onto: left)
                XCTAssertEqual(number(left.label), goal)
            case "more", "tickets":
                let choices = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'counting.ext.choice.'")).allElementsBoundByIndex
                let values = choices.map { number($0.label) }
                let target: Int
                if kind == "more" {
                    let fruits = choices.map { $0.value as! String }
                    XCTAssertEqual(Set(fruits).count, 2)
                    XCTAssertTrue(fruitPairs.insert(fruits.sorted().joined(separator: ":")).inserted)
                    target = ["Find fewer", "Find less treasure"].contains(app.staticTexts["counting.ext.clue"].label) ? values.min()! : values.max()!
                } else { target = number(element("counting.ext.passengers").label) }
                try tap("counting.ext.choice.\(values.first { $0 != target }!)")
                XCTAssertFalse(next.exists)
                try tap("counting.ext.choice.\(target)")
            default: XCTFail("Unknown counting game")
            }
            XCTAssertTrue(next.waitForExistence(timeout: 3), "\(title) level \(index + 1)")
            if index == total - 1 || previewLevels != nil || (["trail", "more"].contains(kind) && index == 0) {
                let shot = XCTAttachment(screenshot: app.screenshot())
                shot.name = title + " level \(index + 1)"; shot.lifetime = .keepAlways; add(shot)
            }
            try reveal(next, in: app); next.tap()
        }
        if previewLevels != nil { app.terminate(); return }
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app); replay.tap()
        XCTAssertTrue(app.staticTexts["counting.ext.level"].waitForExistence(timeout: 5))
        XCTAssertEqual(app.staticTexts["counting.ext.level"].label, "Level 1 of \(total)")
        XCTAssertFalse(app.buttons["counting.ext.next"].exists)
        app.terminate()
    }

            app.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH 'counting.drum.example.'")).allElementsBoundByIndex
                .sorted { $0.identifier < $1.identifier }.map { $0.label.components(separatedBy: ", ").last! }

            if index == 0 {
                app.buttons["counting.drum.watch"].tap()
                waitFor("Watch my pattern", element: phase, timeout: 3)
                XCUIDevice.shared.press(.home); app.activate()
                waitFor("Your turn", element: phase)

    }

        func cell(_ row: Int, _ col: Int) -> XCUIElement { app.buttons["counting.frog.cell.\(row).\(col)"] }
        func route(_ size: Int) throws -> [Int] {
            try (0..<size).map { row in try XCTUnwrap((0..<size).first { col in
                let label = cell(row, col).label
                return label.hasSuffix("frog") || label.hasSuffix("open road") || label.hasSuffix("pond landing")

        }

                let correct = cell(row, path[row])
                XCTAssertGreaterThanOrEqual(correct.frame.width, 44)
                try reveal(correct, in: app); correct.tap()

                XCTAssertEqual(XCTWaiter.wait(for: [hopped], timeout: 5), .completed)
                XCTAssertTrue(correct.label.hasSuffix("frog"))

                    correct.tap(); XCTAssertEqual(progress.label, expected)
                }
            }

        let card = app.buttons["activity-card.Shape Post"]
        XCTAssertTrue(card.waitForExistence(timeout: 5))
        try reveal(card, in: app)
        card.tap()
        XCTAssertTrue(app.buttons["Hear the directions again"].waitForExistence(timeout: 5))
        var seen: Set<String> = []
        for index in 0..<10 {
            let stamp = app.images["shapes.post.stamp"]
            try reveal(stamp, in: app)
            let shape = try XCTUnwrap(stamp.value as? String)
            XCTAssertTrue(seen.insert(shape).inserted)
            let opening = app.otherElements["shapes.post.opening.\(shape)"]
            try reveal(opening, in: app)
            XCTAssertGreaterThanOrEqual(opening.frame.height, 64)
            XCTAssertTrue(stamp.isHittable)
            if index == 0 {
                let screenshot = XCTAttachment(screenshot: app.screenshot())
                screenshot.name = "Shape Post drag board"
                screenshot.lifetime = .keepAlways
                add(screenshot)
                opening.tap()
                XCTAssertFalse(app.buttons["shapes.post.next"].exists, "Tapping must not post a stamp")
                let wrong = app.otherElements.matching(NSPredicate(format: "identifier BEGINSWITH 'shapes.post.opening.' AND identifier != %@", opening.identifier)).firstMatch
                stamp.press(forDuration: 0.1, thenDragTo: wrong)
                XCTAssertFalse(app.buttons["shapes.post.next"].exists)
                stamp.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).press(forDuration: 0.1, thenDragTo: stamp.coordinate(withNormalizedOffset: CGVector(dx: 0.8, dy: 0.5)))
                XCTAssertFalse(app.buttons["shapes.post.next"].exists)
            }
            stamp.press(forDuration: 0.1, thenDragTo: opening)
            let next = app.buttons["shapes.post.next"]
            XCTAssertTrue(next.waitForExistence(timeout: 3))
            try reveal(next, in: app)
            next.tap()
        }
        XCTAssertEqual(seen.count, 10)
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Shape Post completion"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        replay.tap()
        XCTAssertTrue(app.staticTexts["0 of 10 shapes posted"].waitForExistence(timeout: 5))
    }
