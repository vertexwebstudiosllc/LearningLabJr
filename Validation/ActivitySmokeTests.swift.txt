import XCTest
import StoreKitTest

final class ActivitySmokeTests: XCTestCase {
    private struct Category: Decodable { let title: String; let games: [String] }
    private enum NavigationFailure: Error { case missingControl(String) }
    private var session: SKTestSession?

    override func setUpWithError() throws { continueAfterFailure = false }

    @MainActor
    private func launch(premium: Bool = false) async throws -> XCUIApplication {
        let store = try SKTestSession(configurationFileNamed: "LearningLabJr")
        store.resetToDefaultState()
        store.clearTransactions()
        store.disableDialogs = true
        session = store
        if premium { _ = try await store.buyProduct(identifier: "com.learninglabjr.premium.monthly") }
        let app = XCUIApplication()
        app.launchArguments = ["-parents.voicePromptsEnabled", "NO", "-parents.backgroundMusicEnabled", "NO"]
        app.launch()
        XCTAssertTrue(app.buttons["category.ABCs & Phonics"].waitForExistence(timeout: 15))
        return app
    }

    @MainActor
    private func reveal(_ element: XCUIElement, in app: XCUIApplication) throws {
        for _ in 0..<12 {
            let scroll = app.scrollViews.firstMatch
            let viewport = app.windows.firstMatch.frame
            let frame = element.exists ? element.frame : .null
            // XCTest can throw when asked about hittability above the visible window.
            if frame.intersects(viewport) && element.isHittable { return }
            let surface = scroll.exists ? scroll : app
            let movingDown = !frame.isNull && frame.midY < viewport.midY
            let start = surface.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: movingDown ? 0.4 : 0.7))
            let end = surface.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: movingDown ? 0.7 : 0.4))
            start.press(forDuration: 0.05, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.1)
        }
        let hierarchy = XCTAttachment(string: app.debugDescription)
        hierarchy.name = "Navigation failure hierarchy"
        hierarchy.lifetime = .keepAlways
        add(hierarchy)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Navigation failure screen"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        XCTFail("Could not reveal \(element)")
        throw NavigationFailure.missingControl(element.description)
    }

    @MainActor
    private func checkCategory(_ name: String) async throws {
        let app = try await launch(premium: name == "ABCs & Phonics")
        let url = try XCTUnwrap(Bundle(for: Self.self).url(forResource: "ActivityInventory", withExtension: "json"))
        let inventory = try JSONDecoder().decode([Category].self, from: Data(contentsOf: url))
        let category = try XCTUnwrap(inventory.first { $0.title == name })
        let tile = app.buttons["category.\(name)"]
        try reveal(tile, in: app)
        tile.tap()
        let firstCard = app.buttons["activity-card.\(category.games[0])"]
        XCTAssertTrue(firstCard.waitForExistence(timeout: 5))
        try reveal(firstCard, in: app)
        let menuShot = XCTAttachment(screenshot: app.screenshot())
        menuShot.name = name + " menu"
        menuShot.lifetime = .keepAlways
        add(menuShot)
        for (index, title) in category.games.enumerated() {
            let card = app.buttons["activity-card.\(title)"]
            try reveal(card, in: app)
            card.tap()
            if name == "ABCs & Phonics" && (1...6).contains(index) {
                XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists, "Test subscription did not unlock \(title)")
            }
            if name == "Story Time" && index == 0 {
                XCTAssertTrue(app.staticTexts["4 ways to play together"].waitForExistence(timeout: 5))
                let book = app.buttons["activity-card.Owen Onion Finds His Voice"]
                try reveal(book, in: app)
                book.tap()
                XCTAssertTrue(app.buttons["Next page"].waitForExistence(timeout: 5))
                app.buttons["Next page"].tap()
                XCTAssertTrue(app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH 'Page 2 of'")).firstMatch.exists)
                let pageShot = XCTAttachment(screenshot: app.screenshot())
                pageShot.name = "Read Together page two"
                pageShot.lifetime = .keepAlways
                add(pageShot)
                app.navigationBars.buttons.element(boundBy: 0).tap()
            } else {
                XCTAssertTrue(app.buttons["Hear the directions again"].waitForExistence(timeout: 5), title)
                XCTAssertFalse(app.staticTexts["Coming Soon"].exists)
            }
            if index == 0 {
                let screenshot = XCTAttachment(screenshot: app.screenshot())
                screenshot.name = title
                screenshot.lifetime = .keepAlways
                add(screenshot)
            }
            app.navigationBars.buttons.element(boundBy: 0).tap()
        }
        app.terminate()
    }

    @MainActor func testPhonicsRoutes() async throws { try await checkCategory("ABCs & Phonics") }
    @MainActor func testShapesRoutes() async throws { try await checkCategory("Shapes & Colors") }
    @MainActor func testCountingRoutes() async throws { try await checkCategory("123s & Counting") }
    @MainActor func testNatureRoutes() async throws { try await checkCategory("Nature Explorers") }
    @MainActor func testStoryRoutes() async throws { try await checkCategory("Story Time") }
    @MainActor func testFeelingsRoutes() async throws { try await checkCategory("Big Feelings") }

    @MainActor
    func testParentGateRejectsWrongAnswerAndOpensGuide() async throws {
        let app = try await launch(premium: false)
        let parents = app.buttons.matching(NSPredicate(format: "label == 'Parents Corner'")).firstMatch
        try reveal(parents, in: app)
        parents.tap()
        XCTAssertTrue(app.staticTexts["Grown-ups, this part is for you"].waitForExistence(timeout: 5))
        let field = app.textFields["Your answer"]
        field.tap()
        field.typeText("0")
        app.buttons["Continue"].tap()
        XCTAssertTrue(app.staticTexts["That doesn't match yet. Please try again."].exists)
        let question = app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH 'Please add '")).firstMatch.label
        let numbers = question.split(whereSeparator: { !$0.isNumber }).compactMap { Int($0) }
        XCTAssertEqual(numbers.count, 2)
        field.tap()
        field.typeText(String(numbers.reduce(0, +)))
        app.buttons["Continue"].tap()
        XCTAssertTrue(app.staticTexts["Little discoveries, together"].waitForExistence(timeout: 5))
        app.buttons["What each game helps us explore"].tap()
        XCTAssertTrue(app.navigationBars["What we're exploring"].waitForExistence(timeout: 5))
    }

    @MainActor
    func testParentVoicePreviewLeavesSpokenDirectionsOff() async throws {
        let app = try await launch()
        let parents = app.buttons.matching(NSPredicate(format: "label == 'Parents Corner'")).firstMatch
        try reveal(parents, in: app)
        parents.tap()
        let question = app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH 'Please add '")).firstMatch
        XCTAssertTrue(question.waitForExistence(timeout: 5))
        let numbers = question.label.split(whereSeparator: { !$0.isNumber }).compactMap { Int($0) }
        XCTAssertEqual(numbers.count, 2)
        app.textFields["Your answer"].tap()
        app.textFields["Your answer"].typeText(String(numbers.reduce(0, +)))
        app.buttons["Continue"].tap()

        let directions = app.switches.matching(NSPredicate(format: "label CONTAINS 'Spoken directions'")).firstMatch
        try reveal(directions, in: app)
        XCTAssertEqual(directions.value as? String, "0")
        let preview = app.buttons["Hear a friendly sample"]
        try reveal(preview, in: app)
        XCTAssertTrue(preview.isEnabled, "Install an English simulator voice to validate narration preview.")
        preview.tap()
        let stop = app.buttons["Stop voice preview"]
        XCTAssertTrue(stop.waitForExistence(timeout: 3))
        stop.tap()
        XCTAssertTrue(preview.waitForExistence(timeout: 3))
        app.swipeDown()
        XCTAssertEqual(directions.value as? String, "0")
    }

    @MainActor
    func testLandscapeShapePostCompletesAndReplays() async throws {
        let app = try await launch()
        XCUIDevice.shared.orientation = .landscapeLeft
        defer { XCUIDevice.shared.orientation = .portrait }
        let category = app.buttons["category.Shapes & Colors"]
        try reveal(category, in: app)
        category.tap()
        let card = app.buttons["activity-card.Shape Post"]
        XCTAssertTrue(card.waitForExistence(timeout: 5))
        try reveal(card, in: app)
        card.tap()
        XCTAssertTrue(app.buttons["Hear the directions again"].waitForExistence(timeout: 5))
        for shape in ["Circle", "Triangle", "Square", "Triangle", "Circle", "Square"] {
            let opening = app.buttons["shapes.post.opening.\(shape.lowercased())"]
            try reveal(opening, in: app)
            XCTAssertGreaterThanOrEqual(opening.frame.height, 64)
            opening.tap()
        }
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Shape Post landscape completion"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        replay.tap()
        XCTAssertTrue(app.staticTexts["0 of 6 stamps posted"].waitForExistence(timeout: 5))
    }

    @MainActor
    func testTouchCountCompletesAndReplays() async throws {
        let app = try await launch()
        app.buttons["category.123s & Counting"].tap()
        app.buttons["activity-card.Touch & Count"].tap()
        for _ in 0..<3 {
            let next = app.buttons.matching(NSPredicate(format: "label == 'Next adventure' OR label == 'All done'")).firstMatch
            for _ in 0..<6 {
                if next.exists && next.isHittable { break }
                let uncounted = app.buttons.matching(NSPredicate(format: "label CONTAINS 'touch to count'")).firstMatch
                XCTAssertTrue(uncounted.exists, app.debugDescription)
                uncounted.tap()
            }
            try reveal(next, in: app)
            next.tap()
        }
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app)
        replay.tap()
        XCTAssertTrue(app.staticTexts["Adventure 1 of 3"].exists)
    }
}
