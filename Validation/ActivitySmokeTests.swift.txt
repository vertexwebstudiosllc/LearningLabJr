    @MainActor
    private func launch(premium: Bool = false) async throws -> XCUIApplication {
        let store = try SKTestSession(configurationFileNamed: "LearningLabJr")
        store.resetToDefaultState()
        store.clearTransactions()
        store.disableDialogs = true
        session = store
        if premium { _ = try await store.buyProduct(identifier: "com.learninglabjr.premium.monthly") }
        let app = XCUIApplication()
        app.launchArguments = ["-parents.voicePromptsEnabled", "NO", "-parents.backgroundMusicEnabled", "NO", "-parents.useRecordedNarration", "YES", "-validateNarrationCoverage"]
        app.launch()
        XCTAssertTrue(app.buttons["category.ABCs & Phonics"].waitForExistence(timeout: 15))
        return app
    }

    @MainActor
    private func reveal(_ element: XCUIElement, in app: XCUIApplication) throws {
        for _ in 0..<12 {
            let scroll = app.scrollViews.firstMatch
            let viewport = app.windows.firstMatch.frame
            let frame = element.exists ? element.frame : .null
            // XCTest can throw when asked about hittability above the visible window.
            if frame.intersects(viewport) && element.isHittable { return }
            let surface = scroll.exists ? scroll : app
            let movingDown = !frame.isNull && frame.midY < viewport.midY
            let start = surface.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: movingDown ? 0.4 : 0.7))
            let end = surface.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: movingDown ? 0.7 : 0.4))
            start.press(forDuration: 0.05, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.1)
        }
        let hierarchy = XCTAttachment(string: app.debugDescription)
        hierarchy.name = "Navigation failure hierarchy"
        hierarchy.lifetime = .keepAlways
        add(hierarchy)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Navigation failure screen"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        XCTFail("Could not reveal \(element)")
        throw NavigationFailure.missingControl(element.description)
    }

    @MainActor
    private func checkCategory(_ name: String) async throws {
        let app = try await launch(premium: name == "ABCs & Phonics")
        let url = try XCTUnwrap(Bundle(for: Self.self).url(forResource: "ActivityInventory", withExtension: "json"))
        let inventory = try JSONDecoder().decode([Category].self, from: Data(contentsOf: url))
        let category = try XCTUnwrap(inventory.first { $0.title == name })
        let tile = app.buttons["category.\(name)"]
        try reveal(tile, in: app)
        tile.tap()
        let firstCard = app.buttons["activity-card.\(category.games[0])"]
        XCTAssertTrue(firstCard.waitForExistence(timeout: 5))
        try reveal(firstCard, in: app)
        let menuShot = XCTAttachment(screenshot: app.screenshot())
        menuShot.name = name + " menu"
        menuShot.lifetime = .keepAlways
        add(menuShot)
        for (index, title) in category.games.enumerated() {
            let card = app.buttons["activity-card.\(title)"]
            try reveal(card, in: app)
            card.tap()
            if name == "ABCs & Phonics" && (1...6).contains(index) {
                XCTAssertFalse(app.staticTexts["Grown-ups, this part is for you"].exists, "Test subscription did not unlock \(title)")
            }
            if name == "Story Time" && index == 0 {
                XCTAssertTrue(app.staticTexts["4 ways to play together"].waitForExistence(timeout: 5))
                let book = app.buttons["activity-card.Owen Onion Finds His Voice"]
                try reveal(book, in: app)
                book.tap()
                XCTAssertTrue(app.buttons["Next page"].waitForExistence(timeout: 5))
                app.buttons["Next page"].tap()
                XCTAssertTrue(app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH 'Page 2 of'")).firstMatch.exists)
                let pageShot = XCTAttachment(screenshot: app.screenshot())
                pageShot.name = "Read Together page two"
                pageShot.lifetime = .keepAlways
                add(pageShot)
                app.navigationBars.buttons.element(boundBy: 0).tap()
            } else {
                XCTAssertTrue(app.buttons["Hear the directions again"].waitForExistence(timeout: 5), title)
                XCTAssertFalse(app.staticTexts["Coming Soon"].exists)
            }
            if index == 0 {
                let screenshot = XCTAttachment(screenshot: app.screenshot())
                screenshot.name = title
                screenshot.lifetime = .keepAlways
                add(screenshot)
            }
            app.navigationBars.buttons.element(boundBy: 0).tap()
        }
        app.terminate()
    }

    @MainActor
    func testPicnicSixFoodsThreeBasketsAndVariety() async throws {
        let app = try await launch(premium: true)
        app.buttons["category.ABCs & Phonics"].tap()
        let card = app.buttons["activity-card.Picnic Words"]
        try reveal(card, in: app)
        card.tap()
        let baskets = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'picnic.basket.'"))
        let foods = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'picnic.food.'"))
        func list() -> [String] {
            XCTAssertTrue(app.staticTexts["picnic.progress"].waitForExistence(timeout: 5))
            XCTAssertEqual(baskets.count, 3)
            XCTAssertEqual(foods.count, 6)
            return baskets.allElementsBoundByIndex.map(\.label)
        }
        var recent: [String] = []
        for index in 0..<22 {
            let targets = list()
            let key = targets.sorted().joined(separator: ",")
            XCTAssertFalse(recent.suffix(20).contains(key))
            recent.append(key)
            XCTAssertEqual(app.staticTexts["picnic.progress"].label, "0 of 3 foods packed")
            if index == 0 {
                XCTAssertFalse(app.buttons["Previous"].exists)
                XCTAssertFalse(app.buttons["Next food"].exists)
                let viewport = app.windows.firstMatch.frame
                for food in foods.allElementsBoundByIndex {
                    XCTAssertTrue(food.isHittable)
                    XCTAssertTrue(viewport.contains(food.frame), "All six foods must fit on screen")
                }
                let basket = baskets.element(boundBy: 0)
                basket.tap()
                XCTAssertEqual(basket.value as? String, "Empty")
                let wrong = try XCTUnwrap(foods.allElementsBoundByIndex.first { !targets.contains($0.label) })
                wrong.tap()
                XCTAssertEqual(app.staticTexts["picnic.progress"].label, "0 of 3 foods packed")
                XCTAssertFalse(app.buttons["Next picnic"].exists)
            }
            let orders = [[2,0,1], [1,2,0], [0,2,1], [2,1,0], [1,0,2], [0,1,2]]
            for (position, targetIndex) in orders[index % orders.count].enumerated() {
                let word = targets[targetIndex]
                let food = app.buttons["picnic.food.\(word)"]
                try reveal(food, in: app)
                food.tap()
                XCTAssertFalse(food.isEnabled)
                XCTAssertEqual(app.buttons["picnic.basket.\(word)"].value as? String, "Packed")
                XCTAssertEqual(app.staticTexts["picnic.progress"].label, "\(position + 1) of 3 foods packed")
                if position < 2 { XCTAssertFalse(app.buttons["Next picnic"].exists) }
            }
            let next = app.buttons["Next picnic"]
            try reveal(next, in: app)
            next.tap()
        }
        let abandoned = list().sorted().joined(separator: ",")
        XCTAssertFalse(recent.suffix(20).contains(abandoned))
        recent.append(abandoned)
        app.navigationBars.buttons.element(boundBy: 0).tap()
        try reveal(card, in: app)
        card.tap()
        let restarted = list().sorted().joined(separator: ",")
        XCTAssertFalse(recent.suffix(20).contains(restarted))
        recent.append(restarted)
        app.terminate()
        app.launch()
        app.buttons["category.ABCs & Phonics"].tap()
        try reveal(card, in: app)
        card.tap()
        XCTAssertFalse(recent.suffix(20).contains(list().sorted().joined(separator: ",")))
        XCTAssertEqual(app.staticTexts["picnic.progress"].label, "0 of 3 foods packed")
    }

    @MainActor
    func testLetterHideSeekFullAlphabetAndFreshEntry() async throws {
        let app = try await launch()
        app.buttons["category.ABCs & Phonics"].tap()
        let card = app.buttons["activity-card.Letter Hide & Seek"]
        try reveal(card, in: app)
        card.tap()
        let target = app.staticTexts["letter-hide-seek.target"]
        func currentLetter() -> String {
            XCTAssertTrue(target.waitForExistence(timeout: 5))
            return target.label
        }
        let cells = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'letter-hide-seek.cell.'"))
        func matching(_ letter: String) -> [XCUIElement] {
            cells.allElementsBoundByIndex.filter { $0.label.hasPrefix("Letter \(letter),") }
        }
        func complete(last: Bool = false) throws {
            let matches = matching(currentLetter())
            XCTAssertEqual(matches.count, 3)
            for button in matches { try reveal(button, in: app); button.tap() }
            let next = app.buttons[last ? "All done" : "Next letter"]
            try reveal(next, in: app)
            next.tap()
        }
        let initial = currentLetter()
        XCTAssertEqual(cells.count, 6)
        let wrong = try XCTUnwrap(cells.allElementsBoundByIndex.first { !$0.label.hasPrefix("Letter \(initial),") })
        try reveal(wrong, in: app)
        wrong.tap()
        XCTAssertEqual(matching(initial).filter { $0.label.contains(", found,") }.count, 0)
        XCTAssertFalse(app.buttons["Next letter"].exists)
        let first = try XCTUnwrap(matching(initial).first)
        try reveal(first, in: app)
        first.tap()
        first.tap()
        XCTAssertEqual(matching(initial).filter { $0.label.contains(", found,") }.count, 1)
        XCTAssertFalse(app.buttons["Next letter"].exists)
        try complete()
        let abandoned = currentLetter()
        app.navigationBars.buttons.element(boundBy: 0).tap()
        try reveal(card, in: app)
        card.tap()
        let restarted = currentLetter()
        XCTAssertNotEqual(restarted, initial)
        XCTAssertNotEqual(restarted, abandoned)
        var seen: Set<String> = []
        for index in 0..<26 {
            XCTAssertTrue(app.staticTexts["Letter \(index + 1) of 26"].waitForExistence(timeout: 5))
            XCTAssertTrue(seen.insert(currentLetter()).inserted)
            XCTAssertEqual(cells.count, 6)
            try complete(last: index == 25)
        }
        XCTAssertEqual(seen, Set(Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ").map(String.init)))
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app)
        replay.tap()
        XCTAssertNotEqual(currentLetter(), restarted)
        XCTAssertTrue(app.staticTexts["Letter 1 of 26"].exists)
    }

    @MainActor
    func testABCAdventureWholeAlphabetAndNewPictures() async throws {
        let app = try await launch()
        app.buttons["category.ABCs & Phonics"].tap()
        let card = app.buttons["activity-card.ABC Adventure"]
        try reveal(card, in: app)
        card.tap()
        let alphabet = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ").map(String.init)
        var previous: [String: String] = [:]
        for session in 0..<2 {
            for group in 0..<9 {
                let letters = Array(alphabet[(group * 3)..<min(group * 3 + 3, 26)])
                XCTAssertTrue(app.staticTexts["Adventure \(group + 1) of 9"].waitForExistence(timeout: 5))
                XCTAssertTrue(app.staticTexts["0 of \(letters.count) windows open"].exists)
                XCTAssertFalse(app.buttons["Next letters"].exists)
                let windows = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'alphabet-adventure.window.'"))
                XCTAssertEqual(windows.count, letters.count)
                for (index, letter) in letters.enumerated() {
                    let window = app.buttons["alphabet-adventure.window.\(letter)"]
                    try reveal(window, in: app)
                    window.tap()
                    XCTAssertTrue(window.label.hasPrefix("\(letter) is for "))
                    if session == 1 { XCTAssertNotEqual(window.label, previous[letter]) }
                    previous[letter] = window.label
                    if index == 0 {
                        window.tap()
                        XCTAssertTrue(app.staticTexts["1 of \(letters.count) windows open"].exists)
                        XCTAssertFalse(app.buttons["Next letters"].exists)
                    }
                }
                let next = app.buttons[group == 8 ? "All done" : "Next letters"]
                try reveal(next, in: app)
                next.tap()
            }
            let replay = app.buttons["Play again"]
            try reveal(replay, in: app)
            replay.tap()
        }
        // Leaving partway through also creates a fresh session.
        let first = app.buttons["alphabet-adventure.window.A"]
        try reveal(first, in: app)
        first.tap()
        let abandoned = first.label
        app.navigationBars.buttons.element(boundBy: 0).tap()
        try reveal(card, in: app)
        card.tap()
        XCTAssertTrue(app.staticTexts["0 of 3 windows open"].exists)
        try reveal(first, in: app)
        first.tap()
        XCTAssertNotEqual(first.label, abandoned)
    }

    @MainActor
    private func playExpandedCounting(_ kind: String, title: String, total: Int, previewLevels: Int? = nil) async throws {
        let app = try await launch()
        let category = app.buttons["category.123s & Counting"]
        try reveal(category, in: app); category.tap()
        let card = app.buttons["activity-card.\(title)"]
        try reveal(card, in: app); card.tap()
        func element(_ id: String) -> XCUIElement { app.descendants(matching: .any).matching(identifier: id).firstMatch }
        func number(_ label: String) -> Int { label.split(whereSeparator: { !$0.isNumber }).compactMap { Int($0) }.first! }
        func tap(_ id: String) throws { let button = app.buttons[id]; try reveal(button, in: app); button.tap() }
        func listen() throws {
            try tap("counting.ext.listen")
            let drum = app.buttons["counting.ext.tap"]
            let waiting = XCTNSPredicateExpectation(predicate: NSPredicate(format: "enabled == false"), object: drum)
            XCTAssertEqual(XCTWaiter.wait(for: [waiting], timeout: 4), .completed)
            XCTAssertFalse(app.buttons["counting.ext.check"].isEnabled)
            XCTAssertFalse(app.buttons["counting.ext.clear"].isEnabled)
            let ready = XCTNSPredicateExpectation(predicate: NSPredicate(format: "enabled == true"), object: drum)
            XCTAssertEqual(XCTWaiter.wait(for: [ready], timeout: 15), .completed)
        }
        var fruitPairs: Set<String> = []
        for index in 0..<(previewLevels ?? total) {
            let level = app.staticTexts["counting.ext.level"]
            XCTAssertTrue(level.waitForExistence(timeout: 5))
            XCTAssertEqual(level.label, "Level \(index + 1) of \(total)")
            let next = app.buttons["counting.ext.next"]
            XCTAssertFalse(next.exists)
            if previewLevels != nil {
                let shot = XCTAttachment(screenshot: app.screenshot())
                shot.name = title + " start \(index + 1)"; shot.lifetime = .keepAlways; add(shot)
            }
            switch kind {
            case "share":
                let have = number(app.staticTexts["counting.share.have"].label)
                let need = number(app.staticTexts["counting.share.need"].label)
                let target = abs(need - have)
                let choices = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'counting.ext.choice.'")).allElementsBoundByIndex
                XCTAssertEqual(choices.count, 3)
                XCTAssertEqual(Set(choices.map { number($0.label) }).count, 3)
                XCTAssertEqual(app.staticTexts["counting.ext.clue"].label, need > have ? "How many should we add?" : "How many should we take away?")
                try tap(choices.first { number($0.label) != target }!.identifier)
                XCTAssertEqual(level.label, "Level \(index + 1) of \(total)")
                XCTAssertTrue(app.staticTexts["counting.ext.feedback"].exists)
                if previewLevels != nil || index < 2 {
                    let shot = XCTAttachment(screenshot: app.screenshot())
                    shot.name = "Picnic math \(have) to \(need)"; shot.lifetime = .keepAlways; add(shot)
                }
                try tap("counting.ext.choice.\(target)")
                let advanced = XCTNSPredicateExpectation(predicate: NSPredicate(format: "label == %@", "Level \(index + 2) of \(total)"), object: level)
                if index + 1 < total { XCTAssertEqual(XCTWaiter.wait(for: [advanced], timeout: 8), .completed) }
                else { XCTAssertTrue(app.buttons["Play again"].waitForExistence(timeout: 8)) }
                continue
            case "trail":
                let left = element("counting.trail.left")
                let right = element("counting.trail.right")
                let start = number(left.label)
                let goal = number(right.label)
                let target = goal - start
                XCTAssertLessThan(start, goal)
                XCTAssertEqual(start % 5, 0); XCTAssertEqual(goal % 5, 0)
                let chests = app.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH 'counting.trail.chest.'")).allElementsBoundByIndex
                XCTAssertEqual(chests.count, 3)
                let correct = element("counting.trail.chest.\(target)")
                func drop(_ chest: XCUIElement, onto pile: XCUIElement) throws {
                    try reveal(chest, in: app)
                    chest.press(forDuration: 0.15, thenDragTo: pile)
                }
                try drop(correct, onto: right)
                XCTAssertFalse(next.exists); XCTAssertEqual(number(left.label), start)
                let wrong = chests.first { number($0.label) != target }!
                try drop(wrong, onto: left)
                XCTAssertFalse(next.exists); XCTAssertEqual(number(left.label), start)
                try drop(correct, onto: left)
                XCTAssertEqual(number(left.label), goal)
            case "more", "tickets", "dots":
                let choices = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'counting.ext.choice.'")).allElementsBoundByIndex
                let values = choices.map { number($0.label) }
                let target: Int
                if kind == "more" {
                    let fruits = choices.map { $0.value as! String }
                    XCTAssertEqual(Set(fruits).count, 2)
                    XCTAssertTrue(fruitPairs.insert(fruits.sorted().joined(separator: ":")).inserted)
                    target = ["Find fewer", "Find less treasure"].contains(app.staticTexts["counting.ext.clue"].label) ? values.min()! : values.max()!
                } else if kind == "tickets" { target = number(element("counting.ext.passengers").label) }
                else {
                    target = number(element("counting.ext.clue").label)
                    try tap("counting.ext.peek")
                    XCTAssertEqual(element("counting.ext.clue").label, "Hidden clue")
                }
                try tap("counting.ext.choice.\(values.first { $0 != target }!)")
                XCTAssertFalse(next.exists)
                if kind == "dots" {
                    try tap("counting.ext.peek")
                    XCTAssertTrue(element("counting.ext.clue").label.contains("\(target) dots"))
                }
                try tap("counting.ext.choice.\(target)")
            case "towers":
                let target = number(element("counting.ext.model").label)
                var built = number(element("counting.ext.built").label)
                try tap("counting.ext.check"); XCTAssertFalse(next.exists)
                while built != target {
                    try tap(built < target ? "counting.ext.plus" : "counting.ext.minus")
                    built = number(element("counting.ext.built").label)
                }
                try tap("counting.ext.check")
            case "hops":
                let target = number(app.staticTexts["counting.ext.clue"].label)
                try tap("counting.ext.check"); XCTAssertFalse(next.exists)
                try tap("counting.ext.tap"); try tap("counting.ext.clear")
                XCTAssertEqual(app.staticTexts["counting.ext.count"].label, "0 hops")
                for _ in 0..<target { try tap("counting.ext.tap") }
                try tap("counting.ext.check")
            default:
                try listen()
                let target = 1 + index / 3
                XCTAssertEqual(app.staticTexts["counting.ext.count"].label, "Your beats: 0")
                for _ in 0..<target { try tap("counting.ext.tap") }
                try tap("counting.ext.check")
            }
            XCTAssertTrue(next.waitForExistence(timeout: 3), "\(title) level \(index + 1)")
            if index == total - 1 || previewLevels != nil || (["trail", "more"].contains(kind) && index == 0) {
                let shot = XCTAttachment(screenshot: app.screenshot())
                shot.name = title + " level \(index + 1)"; shot.lifetime = .keepAlways; add(shot)
            }
            try reveal(next, in: app); next.tap()
        }
        if previewLevels != nil { app.terminate(); return }
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app); replay.tap()
        XCTAssertTrue(app.staticTexts["counting.ext.level"].waitForExistence(timeout: 5))
        XCTAssertEqual(app.staticTexts["counting.ext.level"].label, "Level 1 of \(total)")
        XCTAssertFalse(app.buttons["counting.ext.next"].exists)
        app.terminate()
    }

        let app = try await launch()

        try reveal(category, in: app); category.tap()
        let card = app.buttons["activity-card.Pattern Train"]
        try reveal(card, in: app); card.tap()
        var seen: Set<String> = []
        for index in 0..<60 {
            let progress = app.staticTexts["shapes.train.progress"]
            XCTAssertTrue(progress.waitForExistence(timeout: 5))
            XCTAssertTrue(progress.label.hasPrefix("Train \(index + 1) of 60"))
            let cars = app.otherElements.matching(NSPredicate(format: "identifier BEGINSWITH 'shapes.train.car.'"))
            let count = cars.count
            XCTAssertTrue((5...15).contains(count))
            let values = (0..<(count - 1)).map { app.otherElements["shapes.train.car.\($0)"].value as! String }
            let period = try XCTUnwrap((2...5).first { size in
                values.count >= 2 * size && values.indices.allSatisfy { values[$0] == values[$0 % size] }
            })
            let key = values.prefix(period).joined(separator: ",")
            XCTAssertTrue(seen.insert(key).inserted)
            let answer = values[(count - 1) % period]
            let choice = app.buttons["shapes.train.choice.\(answer)"]
            try reveal(choice, in: app)
            if index % 12 == 0 {
                let wrong = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH 'shapes.train.choice.' AND identifier != %@", choice.identifier)).firstMatch
                wrong.tap()
                XCTAssertFalse(app.buttons["shapes.train.next"].exists)
                let hint = app.buttons["shapes.train.hint"]
                try reveal(hint, in: app); hint.tap()
            }
            try reveal(choice, in: app); choice.tap()
            let next = app.buttons["shapes.train.next"]
            XCTAssertTrue(next.waitForExistence(timeout: 3))
            XCTAssertFalse(choice.isEnabled)
            XCTAssertEqual(app.otherElements["shapes.train.car.\(count - 1)"].value as? String, answer)
            if index == 59 {
                let board = app.otherElements["shapes.train.car.0"]
                try reveal(board, in: app)
                let shot = XCTAttachment(screenshot: app.screenshot())
                shot.name = "Pattern Train fifteen shapes"
                shot.lifetime = .keepAlways; add(shot)
            }
            try reveal(next, in: app); next.tap()
        }
        XCTAssertEqual(seen.count, 60)
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app); replay.tap()
        XCTAssertTrue(app.staticTexts["shapes.train.progress"].waitForExistence(timeout: 5))
        XCTAssertTrue(app.staticTexts["shapes.train.progress"].label.hasPrefix("Train 1 of 60"))
    }

    @MainActor
    func testColorLaundryCyclesAndReentry() async throws {
        let app = try await launch()
        let category = app.buttons["category.Shapes & Colors"]
        try reveal(category, in: app)
        category.tap()
        let card = app.buttons["activity-card.Color Laundry"]
        try reveal(card, in: app)
        card.tap()
        XCTAssertTrue(app.images["shapes.laundry.shirt"].waitForExistence(timeout: 5))
        var previous: String?
        for _ in 0..<2 {
            var seen: Set<String> = []
            for index in 0..<11 {
                let shirt = app.images["shapes.laundry.shirt"]
                let color = try XCTUnwrap(shirt.value as? String)
                if index == 0 { XCTAssertNotEqual(color, previous) }
                XCTAssertTrue(seen.insert(color).inserted)
                let basket = app.otherElements["shapes.laundry.basket.\(color)"]
                try reveal(basket, in: app)
                let choices = app.otherElements.matching(NSPredicate(format: "identifier BEGINSWITH 'shapes.laundry.basket.'"))
                XCTAssertEqual(choices.count, 3)
                let wrong = choices.allElementsBoundByIndex.first { $0.identifier != basket.identifier }!
                basket.tap()
                XCTAssertFalse(app.buttons["shapes.laundry.next"].exists, "Taps must not sort shirts")
                shirt.press(forDuration: 0.1, thenDragTo: wrong)
                XCTAssertFalse(app.buttons["shapes.laundry.next"].exists)
                if index == 0 {
                    shirt.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).press(forDuration: 0.1, thenDragTo: shirt.coordinate(withNormalizedOffset: CGVector(dx: 0.8, dy: 0.5)))
                    XCTAssertFalse(app.buttons["shapes.laundry.next"].exists)
                }
                if color == "pink" || color == "white" {
                    let shot = XCTAttachment(screenshot: app.screenshot())
                    shot.name = "Color Laundry \(color) drag board"
                    shot.lifetime = .keepAlways
                    add(shot)
                }
                shirt.press(forDuration: 0.1, thenDragTo: basket)
                let next = app.buttons["shapes.laundry.next"]
                XCTAssertTrue(next.waitForExistence(timeout: 3))
                let progress = app.staticTexts["shapes.laundry.progress"].label
                shirt.press(forDuration: 0.1, thenDragTo: basket)
                XCTAssertEqual(app.staticTexts["shapes.laundry.progress"].label, progress)
                try reveal(next, in: app)
                next.tap()
                previous = color
            }
            XCTAssertEqual(seen.count, 11)
        }
        app.buttons["BackButton"].tap()
        try reveal(card, in: app)
        card.tap()
        XCTAssertTrue(app.staticTexts["0 shirts sorted"].waitForExistence(timeout: 5))
    }

    @MainActor
    private func checkShapePost(orientation: UIDeviceOrientation) async throws {
        let app = try await launch()
        XCUIDevice.shared.orientation = orientation
        defer { XCUIDevice.shared.orientation = .portrait }
        let category = app.buttons["category.Shapes & Colors"]
        try reveal(category, in: app)
        category.tap()
        let card = app.buttons["activity-card.Shape Post"]
        XCTAssertTrue(card.waitForExistence(timeout: 5))
        try reveal(card, in: app)
        card.tap()
        XCTAssertTrue(app.buttons["Hear the directions again"].waitForExistence(timeout: 5))
        var seen: Set<String> = []
        for index in 0..<10 {
            let stamp = app.images["shapes.post.stamp"]
            try reveal(stamp, in: app)
            let shape = try XCTUnwrap(stamp.value as? String)
            XCTAssertTrue(seen.insert(shape).inserted)
            let opening = app.otherElements["shapes.post.opening.\(shape)"]
            try reveal(opening, in: app)
            XCTAssertGreaterThanOrEqual(opening.frame.height, 64)
            XCTAssertTrue(stamp.isHittable)
            if index == 0 {
                let screenshot = XCTAttachment(screenshot: app.screenshot())
                screenshot.name = "Shape Post drag board"
                screenshot.lifetime = .keepAlways
                add(screenshot)
                opening.tap()
                XCTAssertFalse(app.buttons["shapes.post.next"].exists, "Tapping must not post a stamp")
                let wrong = app.otherElements.matching(NSPredicate(format: "identifier BEGINSWITH 'shapes.post.opening.' AND identifier != %@", opening.identifier)).firstMatch
                stamp.press(forDuration: 0.1, thenDragTo: wrong)
                XCTAssertFalse(app.buttons["shapes.post.next"].exists)
                stamp.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).press(forDuration: 0.1, thenDragTo: stamp.coordinate(withNormalizedOffset: CGVector(dx: 0.8, dy: 0.5)))
                XCTAssertFalse(app.buttons["shapes.post.next"].exists)
            }
            stamp.press(forDuration: 0.1, thenDragTo: opening)
            let next = app.buttons["shapes.post.next"]
            XCTAssertTrue(next.waitForExistence(timeout: 3))
            try reveal(next, in: app)
            next.tap()
        }
        XCTAssertEqual(seen.count, 10)
        let replay = app.buttons["Play again"]
        try reveal(replay, in: app)
        let screenshot = XCTAttachment(screenshot: app.screenshot())
        screenshot.name = "Shape Post completion"
        screenshot.lifetime = .keepAlways
        add(screenshot)
        replay.tap()
        XCTAssertTrue(app.staticTexts["0 of 10 shapes posted"].waitForExistence(timeout: 5))
    }
